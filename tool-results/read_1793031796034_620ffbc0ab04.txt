   120→  let x = Math.abs(a - b) % (Math.PI * 2)

   215→  private buildColliders() {
   216→    this.colliders = [{ x: LAKE.x, y: LAKE.y, r: LAKE.r - 5, propId: 'lake' }]
   217→    for (const p of this.props) {
   218→      if (!p.solid) continue
   219→      const r =
   220→        p.kind === 'house'
   221→          ? Math.max(p.w ?? 100, p.h ?? 60) * 0.36
   222→          : p.kind === 'tent'
   223→            ? (p.w ?? 60) * 0.4
   224→            : (p.r ?? 10)
   225→      this.colliders.push({ x: p.pos.x, y: p.pos.y, r, propId: p.id })
   226→    }
   227→  }

   247→  private recalcStats(full = false) {
   248→    const newMaxHp = 80 + this.attributes.vigor * 8 + this.skills.vitality * 15 + (this.level - 1) * 8
   249→    if (full) this.maxHp = newMaxHp
   250→    else if (newMaxHp !== this.maxHp) {
   251→      const ratio = this.hp / this.maxHp
   252→      this.maxHp = newMaxHp
   253→      this.hp = Math.min(this.maxHp, Math.max(1, Math.round(ratio * this.maxHp)))
   254→    }
   255→    this.maxStamina = 70 + this.attributes.vigor * 3 + (this.level - 1) * 4
   256→    this.stamina = Math.min(this.stamina, this.maxStamina)
   257→    this.refreshPlayerStats()
   258→  }
   259→
   260→  private refreshPlayerStats() {
   261→    if (!this.player) return
   262→    this.player.maxHp = this.maxHp
   263→    this.player.hp = Math.max(0, Math.min(this.player.hp, this.maxHp))
   264→    this.player.atkBonus = this.playerAtk().atkBonus
   265→    this.player.ac = this.playerAc()
   266→    this.player.dmgMod = this.playerAtk().mod
   267→  }

   273→      mod: (w?.damage ?? 2) + Math.floor(this.attributes.might / 2),
   274→    }
   275→  }

   370→  private spawnWorld() {
   371→    for (const n of NPCS) {
   372→      if (n.id === 'kaya' && this.companionIds.kaya) continue
   373→      if (n.id === 'torvald_npc' && this.companionIds.torvald) continue
   374→      this.entities.push(
   375→        this.makeEntity({
   376→          id: n.id,
   377→          kind: 'npc',
   378→          name: n.name,
   379→          role: n.role,
   380→          pos: { ...n.pos },
   381→          color: n.color,
   382→          radius: 13,
   383→          hp: 1,
   384→          maxHp: 1,
   385→          team: 'village',
   386→          speed: 0,
   387→          npc: n,
   388→        }),
   389→      )
   390→    }
   391→    for (const z of SPAWN_ZONES) {
   392→      for (let i = 0; i < z.max; i++) this.spawnZoneEnemy(z)
   393→    }
   394→    for (const u of UNIQUE_SPAWNS) {
   395→      if (u.enemy === 'alpha' && this.flags.alphaDead) continue
   396→      if (u.enemy === 'captain' && this.flags.captainDead) continue
   397→      if (u.enemy === 'boss' && this.flags.bossDead) continue
   398→      this.spawnUnique(u.enemy, u.pos)
   399→    }
   400→    if (this.companionIds.kaya) this.spawnCompanion('kaya')
   401→    if (this.companionIds.torvald) this.spawnCompanion('torvald')
   402→    if (this.flags.gateOpen) {
   403→      this.removedProps.add('barrier')
   404→      this.removedProps.add('barrier2')
   405→      this.removedProps.add('barrier3')
   406→    }
   407→    for (let i = 1; i <= 4; i++) {
   408→      if (this.flags['herb_herb' + i]) this.removedProps.add('herb' + i)
   409→    }
   410→  }

   419→      name: ENEMY_NAME[z.enemy],
   420→      pos,
   421→      radius: st.radius,
   422→      hp: st.hp,
   423→      maxHp: st.hp,
   424→      speed: st.speed,

   505→  private onKeyDown = (ev: KeyboardEvent) => {
   506→    const store = useGame.getState()
   507→    if (this.dialogueTree) {
   508→      const n = parseInt(ev.key, 10)
   509→      if (!isNaN(n) && n >= 1 && n <= 9) {
   510→        this.chooseOption(n - 1)
   511→        ev.preventDefault()
   512→      } else if (ev.key === 'Escape') {
   513→        this.closeDialogue()
   514→      }
   515→      return
   516→    }
   517→    if (ev.key === 'Escape') {
   518→      if (store.panel && store.panel !== 'ending') this.closePanel()
   519→      else if (!store.panel) this.openPanel('pause')
   520→      return
   521→    }
   522→    if (store.panel) {
   523→      const map: Record<string, PanelId> = { c: 'char', i: 'inv', j: 'journal', m: 'map' }
   524→      const k = ev.key.toLowerCase()
   525→      if (map[k] && store.panel === map[k]) {
   526→        this.closePanel()
   527→        ev.preventDefault()
   528→      }
   529→      return
   530→    }
   531→    const k = ev.key.toLowerCase()
   532→    this.keys.add(k)
   533→    if (k === ' ') {
   534→      this.tacticalPaused = !this.tacticalPaused
   535→      useGame.getState().patch({ paused: this.tacticalPaused })
   536→      sfx('click')
   537→      ev.preventDefault()
   538→      return
   539→    }
   540→    if (k === 'f' || k === 'а') this.doInteract()
   541→    if (k === 'c' || k === 'с') this.openPanel('char')
   542→    if (k === 'i' || k === 'ш') this.openPanel('inv')
   543→    if (k === 'j' || k === 'о') this.openPanel('journal')

   670→  private update(dt: number) {

   687→    this.camera.y += (this.player.pos.y - this.camera.y) * lerp
   688→    // не показываем зоны за пределами мира

   691→    this.camera.x = Math.max(Math.min(vw / 2, WORLD.w / 2), Math.min(this.camera.x, WORLD.w - vw / 2))

   721→  private moveWithCollisions(e: Entity, vx: number, vy: number, dt: number) {
   722→    const kbi = this.kb.get(e.eid)
   723→    let nx = e.pos.x + (vx + (kbi?.x ?? 0)) * dt
   724→    let ny = e.pos.y + (vy + (kbi?.y ?? 0)) * dt
   725→    if (kbi) {
   726→      const decay = Math.pow(0.002, dt)
   727→      kbi.x *= decay
   728→      kbi.y *= decay
   729→      if (Math.hypot(kbi.x, kbi.y) < 8) this.kb.delete(e.eid)
   730→    }
   731→    nx = Math.max(30, Math.min(WORLD.w - 30, nx))
   732→    ny = Math.max(30, Math.min(WORLD.h - 30, ny))
   733→    for (const c of this.colliders) {
   734→      if (c.propId && this.removedProps.has(c.propId)) continue
   735→      const dx = nx - c.x
   736→      const dy = ny - c.y
   737→      const dist = Math.hypot(dx, dy)
   738→      const min = c.r + e.radius * 0.7
   739→      if (dist < min && dist > 0.001) {
   740→        nx = c.x + (dx / dist) * min
   741→        ny = c.y + (dy / dist) * min
   742→      }
   743→    }
   744→    e.pos.x = nx
   745→    e.pos.y = ny
   746→  }

   748→  private separate() {
   749→    const alive = this.entities.filter((e) => e.alive && e.kind !== 'npc')
   750→    for (let i = 0; i < alive.length; i++) {
   751→      for (let j = i + 1; j < alive.length; j++) {
   752→        const a = alive[i]
   753→        const b = alive[j]
   754→        const dx = b.pos.x - a.pos.x
   755→        const dy = b.pos.y - a.pos.y
   756→        const dist = Math.hypot(dx, dy)
   757→        const min = a.radius + b.radius
   758→        if (dist < min && dist > 0.01) {
   759→          const push = ((min - dist) / 2) * 0.6
   760→          const ux = dx / dist
   761→          const uy = dy / dist
   762→          if (a.kind !== 'player') {
   763→            a.pos.x -= ux * push
   764→            a.pos.y -= uy * push
   765→          }
   766→          if (b.kind !== 'player') {
   767→            b.pos.x += ux * push
   768→            b.pos.y += uy * push
   769→          }
   770→        }
   771→      }
   772→    }
   773→  }

   803→  private updatePlayer(dt: number) {
   804→    const p = this.player
   805→    if (!p.alive) return
   806→    let mx = 0
   807→    let my = 0
   808→    if (this.keys.has('w') || this.keys.has('arrowup') || this.keys.has('ц')) my -= 1
   809→    if (this.keys.has('s') || this.keys.has('arrowdown') || this.keys.has('ы')) my += 1
   810→    if (this.keys.has('a') || this.keys.has('arrowleft') || this.keys.has('ф')) mx -= 1
   811→    if (this.keys.has('d') || this.keys.has('arrowright') || this.keys.has('в')) mx += 1
   812→    const moving = mx !== 0 || my !== 0
   813→    if (moving) {
   814→      const l = Math.hypot(mx, my)
   815→      p.vel = { x: (mx / l) * p.speed, y: (my / l) * p.speed }
   816→      p.dest = null
   817→    } else if (p.dest) {
   818→      const dx = p.dest.x - p.pos.x
   819→      const dy = p.dest.y - p.pos.y
   820→      const dist = Math.hypot(dx, dy)
   821→      if (dist < 8) p.dest = null
   822→      else p.vel = { x: (dx / dist) * p.speed, y: (dy / dist) * p.speed }
   823→    } else {
   824→      p.vel = { x: 0, y: 0 }
   825→    }
   826→    const target = this.targetEid ? this.entities.find((e) => e.eid === this.targetEid) : null
   827→    if (target && target.alive) {
   828→      p.facing = Math.atan2(target.pos.y - p.pos.y, target.pos.x - p.pos.x)
   829→      const dist = Math.hypot(target.pos.x - p.pos.x, target.pos.y - p.pos.y)
   830→      if (dist > p.attackRange + target.radius * 0.6 && !moving) {
   831→        p.dest = { ...target.pos }
   832→      } else if (dist <= p.attackRange + target.radius * 0.6) {
   833→        p.dest = null
   834→        if (p.cds['atk'] <= 0) this.meleeAttack(p, target)
   835→      }
   836→    } else if (moving) {
   837→      p.facing = Math.atan2(p.vel.y, p.vel.x)
   838→    }
   839→    this.moveWithCollisions(p, p.vel.x, p.vel.y, dt)
   840→    p.walkPhase += dt * (Math.hypot(p.vel.x, p.vel.y) > 4 ? 1 : 0.2)
   841→    const inCombat = this.nearestEnemyOf(p, 320) !== null
   842→    this.stamina = Math.min(this.maxStamina, this.stamina + (inCombat ? 6 : 14) * dt)
   843→  }

   845→  private updateCompanion(e: Entity, dt: number) {
   846→    if (!e.alive) {
   847→      e.downT -= dt
   848→      if (e.downT <= 0) {
   849→        e.alive = true
   850→        e.hp = Math.round(e.maxHp * 0.45)
   851→        this.floater(e.pos, 'На ноги!', '#8fbf7f', 14)
   852→      }
   853→      return
   854→    }
   855→    const stance: Stance = e.stance ?? 'follow'
   856→    const player = this.player
   857→    let desired: Vec | null = null
   858→    let target: Entity | null = null
   859→
   860→    if (stance === 'hold') {
   861→      if (Math.hypot(e.home.x - e.pos.x, e.home.y - e.pos.y) > 14) desired = { ...e.home }
   862→      target = this.nearestEnemyOf(e, 260)
   863→    } else if (stance === 'aggressive') {
   864→      target = this.nearestEnemyOf(e, 460)
   865→      if (!target) desired = { x: player.pos.x - 20, y: player.pos.y + 16 }
   866→    } else {
   867→      const t = this.targetEid ? this.entities.find((x) => x.eid === this.targetEid) : null
   868→      if (t && t.alive && Math.hypot(t.pos.x - e.pos.x, t.pos.y - e.pos.y) < 400) target = t
   869→      else target = this.nearestEnemyOf(e, 240)
   870→      const off = e.id === 'torvald' ? { x: -30, y: 22 } : { x: -26, y: -26 }
   871→      desired = { x: player.pos.x + off.x, y: player.pos.y + off.y }
   872→    }
   873→    if (target) {
   874→      const dist = Math.hypot(target.pos.x - e.pos.x, target.pos.y - e.pos.y)
   875→      const stopAt = e.ranged ? e.ranged * 0.75 : e.attackRange + target.radius * 0.6
   876→      desired = dist > stopAt ? { ...target.pos } : null
   877→      e.facing = Math.atan2(target.pos.y - e.pos.y, target.pos.x - e.pos.x)
   878→      if (dist <= (e.ranged ?? e.attackRange) + target.radius * 0.6 && e.cds['atk'] <= 0) {
   879→        if (e.ranged) this.rangedAttack(e, target)
   880→        else this.meleeAttack(e, target)
   881→      }
   882→      if (e.id === 'torvald') {
   883→        if (e.cds['shieldbash'] <= 0 && dist < 70) this.castCompanionAbility(e.eid, 'shieldbash', target.eid)
   884→        if (e.cds['warcry'] <= 0 && dist < 300) this.castCompanionAbility(e.eid, 'warcry', target.eid)
   885→      }
   886→      if (e.id === 'kaya' && e.cds['frostbolt'] <= 0 && dist < 260) {
   887→        this.castCompanionAbility(e.eid, 'frostbolt', target.eid)
   888→      }
   889→    }
   890→    if (e.id === 'kaya' && e.cds['mend'] <= 0) this.castCompanionAbility(e.eid, 'mend', null)
   891→
   892→    if (desired) {
   893→      const dx = desired.x - e.pos.x
   894→      const dy = desired.y - e.pos.y
   895→      const dist = Math.hypot(dx, dy)
   896→      if (dist > 10) {
   897→        const sp = e.speed * (e.slowT > 0 ? 0.5 : 1)
   898→        e.vel = { x: (dx / dist) * sp, y: (dy / dist) * sp }
   899→      } else e.vel = { x: 0, y: 0 }
   900→    } else {
   901→      e.vel = { x: 0, y: 0 }
   902→    }
   903→    if (!target && Math.hypot(e.vel.x, e.vel.y) > 4) {
   904→      e.facing = Math.atan2(e.vel.y, e.vel.x)
   905→    }
   906→    this.moveWithCollisions(e, e.vel.x, e.vel.y, dt)
   907→    e.walkPhase += dt * (Math.hypot(e.vel.x, e.vel.y) > 4 ? 1 : 0.2)
   908→  }

   969→  private updateEnemy(e: Entity, dt: number) {
   970→    if (!e.alive) {
   971→      if (e.respawnAt !== undefined && this.time >= e.respawnAt) {
   972→        if (Math.hypot(this.player.pos.x - e.home.x, this.player.pos.y - e.home.y) > 500) {
   973→          e.alive = true
   974→          e.hp = e.maxHp
   975→          e.aiState = 'idle'
   976→          e.burnT = 0
   977→          e.stunT = 0
   978→          const zone = SPAWN_ZONES.find((z) => z.id === e.zoneId)
   979→          if (zone) {
   980→            const a = Math.random() * Math.PI * 2
   981→            const rr = Math.random() * zone.radius
   982→            e.pos = { x: zone.center.x + Math.cos(a) * rr, y: zone.center.y + Math.sin(a) * rr }
   983→            e.home = { ...e.pos }
   984→          }
   985→          e.respawnAt = undefined
   986→        }
   987→      }
   988→      return
   989→    }
   990→    // Интро босса
   991→    if (e.boss && !this.flags.bossIntroDone) {
   992→      if (Math.hypot(this.player.pos.x - e.pos.x, this.player.pos.y - e.pos.y) < 340) {
   993→        this.flags.bossIntroDone = true
   994→        e.aggroR = 400
   995→        this.openDialogue('boss_intro')
   996→        return
   997→      }
   998→    }
   999→    if (e.stunT > 0) {
  1000→      e.vel = { x: 0, y: 0 }
  1001→      this.moveWithCollisions(e, 0, 0, dt)
  1002→      return
  1003→    }
  1004→    const prey = this.nearestParty(e.pos, e.aiState === 'idle' || e.aiState === 'wander' ? e.aggroR : 1e9)
  1005→    const homeDist = Math.hypot(e.pos.x - e.home.x, e.pos.y - e.home.y)
  1006→
  1007→    // поводок: слишком далеко от дома — возвращаемся

  1009→      e.aiState = 'return'

  1011→

  1020→        e.vel = { x: (dx / l) * e.speed, y: (dy / l) * e.speed }
  1021→        e.facing = Math.atan2(dy, dx)
  1022→        this.moveWithCollisions(e, e.vel.x, e.vel.y, dt)
  1023→        e.hp = Math.min(e.maxHp, e.hp + e.maxHp * 0.12 * dt)
  1024→        e.walkPhase += dt
  1025→      }
  1026→      return
  1027→    }
  1028→
  1029→    if (!prey) {
  1030→      if (e.aiState === 'chase' || e.aiState === 'attack') {
  1031→        e.aiState = 'return'
  1032→      } else {
  1033→        e.wanderT -= dt
  1034→        if (e.wanderT <= 0) {
  1035→          e.wanderT = rndRange(1.5, 4)
  1036→          const a = Math.random() * Math.PI * 2
  1037→          const rr = rndRange(0, 60)
  1038→          e.dest = { x: e.home.x + Math.cos(a) * rr, y: e.home.y + Math.sin(a) * rr }

  1042→          const dy = e.dest.y - e.pos.y
  1043→          const l = Math.hypot(dx, dy)
  1044→          if (l < 6) e.dest = null

  1046→            e.vel = { x: (dx / l) * e.speed * 0.35, y: (dy / l) * e.speed * 0.35 }
  1047→            e.facing = Math.atan2(e.vel.y, e.vel.x)
  1048→            this.moveWithCollisions(e, e.vel.x, e.vel.y, dt)
  1049→            e.walkPhase += dt
  1050→            return
  1051→          }
  1052→        }
  1053→        e.vel = { x: 0, y: 0 }
  1054→      }
  1055→      return
  1056→    }
  1057→
  1058→    e.aiState = 'chase'
  1059→    const dist = Math.hypot(prey.pos.x - e.pos.x, prey.pos.y - e.pos.y)
  1060→    e.facing = Math.atan2(prey.pos.y - e.pos.y, prey.pos.x - e.pos.x)
  1061→
  1062→    if (e.boss) this.bossLogic(e, dist)
  1063→
  1064→    const isRanged = !!e.ranged
  1065→    const wantRange = isRanged ? e.ranged! * 0.85 : e.attackRange + prey.radius * 0.6
  1066→    let vx = 0
  1067→    let vy = 0
  1068→    if (isRanged && dist < wantRange * 0.6) {
  1069→      vx = -Math.cos(e.facing) * e.speed
  1070→      vy = -Math.sin(e.facing) * e.speed
  1071→    } else if (dist > wantRange) {
  1072→      vx = Math.cos(e.facing) * e.speed
  1073→      vy = Math.sin(e.facing) * e.speed
  1074→    }
  1075→    const spd = e.slowT > 0 ? 0.5 : 1
  1076→    this.moveWithCollisions(e, vx * spd, vy * spd, dt)
  1077→    e.walkPhase += dt * (Math.hypot(vx, vy) > 4 ? 1 : 0.2)
  1078→
  1079→    if (e.cds['atk'] <= 0) {
  1080→      if (isRanged && dist < e.ranged!) this.rangedAttack(e, prey)
  1081→      else if (!isRanged && dist <= e.attackRange + prey.radius * 0.6) this.meleeAttack(e, prey)
  1082→    }
  1083→  }

  1538→  private evalToken(s: string): boolean {
  1539→    let m = s.match(/^quest\.(\w+)==(\d+)$/)
  1540→    if (m) return this.quests[m[1]].stage === parseInt(m[2], 10)
  1541→    m = s.match(/^quest\.(\w+)>=(\d+)$/)
  1542→    if (m) return this.quests[m[1]].stage >= parseInt(m[2], 10)
  1543→    m = s.match(/^quest\.(\w+)<(\d+)$/)
  1544→    if (m) return this.quests[m[1]].stage < parseInt(m[2], 10)
  1545→    m = s.match(/^flag\.(\w+)$/)
  1546→    if (m) return !!this.flags[m[1]]
  1547→    m = s.match(/^!flag\.(\w+)$/)
  1548→    if (m) return !this.flags[m[1]]
  1549→    m = s.match(/^faction==(\w+)$/)
  1550→    if (m) return this.faction === (m[1] as Faction)
  1551→    m = s.match(/^comp\.(\w+)$/)
  1552→    if (m) return !!this.companionIds[m[1] as 'kaya' | 'torvald']
  1553→    m = s.match(/^!comp\.(\w+)$/)
  1554→    if (m) return !this.companionIds[m[1] as 'kaya' | 'torvald']
  1555→    m = s.match(/^has\.(\w+)(?::(\d+))?$/)
  1556→    if (m) return this.countItem(m[1]) >= (m[2] ? parseInt(m[2], 10) : 1)
  1557→    m = s.match(/^lp>=(\d+)$/)
  1558→    if (m) return this.lp >= parseInt(m[1], 10)
  1559→    m = s.match(/^gold>=(\d+)$/)
  1560→    if (m) return this.gold >= parseInt(m[1], 10)
  1561→    return true
  1562→  }
