   215→   215→  private buildColliders() {
   216→   216→    this.colliders = [{ x: LAKE.x, y: LAKE.y, r: LAKE.r - 5, propId: 'lake' }]

   220→   220→        p.kind === 'house'
   221→   221→          ? Math.max(p.w ?? 100, p.h ?? 60) * 0.36
   222→   222→          : p.kind === 'tent'

   247→   247→  private recalcStats(full = false) {
   248→   248→    const newMaxHp = 80 + this.attributes.vigor * 8 + this.skills.vitality * 15 + (this.level - 1) * 8
   249→   249→    if (full) this.maxHp = newMaxHp
   250→   250→    else if (newMaxHp !== this.maxHp) {
   251→   251→      const ratio = this.hp / this.maxHp
   252→   252→      this.maxHp = newMaxHp
   253→   253→      this.hp = Math.min(this.maxHp, Math.max(1, Math.round(ratio * this.maxHp)))
   254→   254→    }
   255→   255→    this.maxStamina = 70 + this.attributes.vigor * 3 + (this.level - 1) * 4
   256→   256→    this.stamina = Math.min(this.stamina, this.maxStamina)
   257→   257→    this.refreshPlayerStats()
   258→   258→  }

   375→   375→        this.makeEntity({

   377→   377→          kind: 'npc',
   378→   378→          name: n.name,

   386→   386→          speed: 0,
   387→   387→          npc: n,
   388→   388→        }),
   389→   389→      )
   390→   390→    }
   391→   391→    for (const z of SPAWN_ZONES) {
   392→   392→      for (let i = 0; i < z.max; i++) this.spawnZoneEnemy(z)
   393→   393→    }
   394→   394→    for (const u of UNIQUE_SPAWNS) {
   395→   395→      if (u.enemy === 'alpha' && this.flags.alphaDead) continue
   396→   396→      if (u.enemy === 'captain' && this.flags.captainDead) continue
   397→   397→      if (u.enemy === 'boss' && this.flags.bossDead) continue
   398→   398→      this.spawnUnique(u.enemy, u.pos)
   399→   399→    }
   400→   400→    if (this.companionIds.kaya) this.spawnCompanion('kaya')
   401→   401→    if (this.companionIds.torvald) this.spawnCompanion('torvald')
   402→   402→    if (this.flags.gateOpen) {
   403→   403→      this.removedProps.add('barrier')
   404→   404→      this.removedProps.add('barrier2')
   405→   405→      this.removedProps.add('barrier3')
   406→   406→    }
   407→   407→    for (let i = 1; i <= 4; i++) {
   408→   408→      if (this.flags['herb_herb' + i]) this.removedProps.add('herb' + i)
   409→   409→    }
   410→   410→  }
