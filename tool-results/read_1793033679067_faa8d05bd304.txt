     4→// ---------------- Терен (запекается один раз) ----------------
     5→
     6→export function bakeTerrain(): HTMLCanvasElement {
     7→  const c = document.createElement('canvas')
     8→  c.width = WORLD.w
     9→  c.height = WORLD.h
    10→  const ctx = c.getContext('2d') as CanvasRenderingContext2D
    11→
    12→  // База — тёмная северная тайга
    13→  ctx.fillStyle = '#39472c'
    14→  ctx.fillRect(0, 0, WORLD.w, WORLD.h)
    15→
    16→  // Шумовые пятна
    17→  let seed = 99
    18→  const rnd = () => {
    19→    seed = (seed * 16807) % 2147483647
    20→    return seed / 2147483647
    21→  }
    22→  for (let i = 0; i < 2600; i++) {
    23→    const x = rnd() * WORLD.w
    24→    const y = rnd() * WORLD.h
    25→    const r = 14 + rnd() * 60
    26→    ctx.fillStyle = rnd() > 0.5 ? 'rgba(47,58,36,0.35)' : 'rgba(72,88,48,0.25)'
    27→    ctx.beginPath()
    28→    ctx.arc(x, y, r, 0, Math.PI * 2)
    29→    ctx.fill()
    30→  }
    31→  // Снежная дымка на севере
    32→  for (let i = 0; i < 500; i++) {
    33→    const x = rnd() * WORLD.w
    34→    const y = rnd() * 700
    35→    ctx.fillStyle = 'rgba(220,225,215,0.05)'
    36→    ctx.beginPath()
    37→    ctx.arc(x, y, 20 + rnd() * 70, 0, Math.PI * 2)
    38→    ctx.fill()
    39→  }
    40→
    41→  // Зона Разлома — пепел и трещины
    42→  const rg = ctx.createRadialGradient(2950, 280, 60, 2950, 280, 420)
    43→  rg.addColorStop(0, 'rgba(74,58,80,0.95)')
    44→  rg.addColorStop(0.6, 'rgba(64,50,70,0.6)')
    45→  rg.addColorStop(1, 'rgba(64,50,70,0)')
    46→  ctx.fillStyle = rg
    47→  ctx.fillRect(2500, -140, 800, 840)
    48→  for (let i = 0; i < 26; i++) {
    49→    const x = 2720 + rnd() * 480
    50→    const y = 100 + rnd() * 420
    51→    ctx.strokeStyle = 'rgba(150,100,190,0.30)'
    52→    ctx.lineWidth = 1 + rnd() * 2
    53→    ctx.beginPath()
    54→    ctx.moveTo(x, y)
    55→    ctx.lineTo(x + (rnd() - 0.5) * 70, y + (rnd() - 0.5) * 70)
    56→    ctx.lineTo(x + (rnd() - 0.5) * 110, y + (rnd() - 0.5) * 110)
    57→    ctx.stroke()
    58→  }
    59→
    60→  // Дороги
    61→  const drawRoad = (pts: { x: number; y: number }[], w1: number, color: string) => {
    62→    ctx.strokeStyle = color
    63→    ctx.lineWidth = w1
    64→    ctx.lineCap = 'round'
    65→    ctx.lineJoin = 'round'
    66→    ctx.beginPath()
    67→    pts.forEach((p, i) => (i === 0 ? ctx.moveTo(p.x, p.y) : ctx.lineTo(p.x, p.y)))
    68→    ctx.stroke()
    69→  }
    70→  for (const road of ROADS) drawRoad(road, 40, 'rgba(96,78,52,0.85)')
    71→  for (const road of ROADS) drawRoad(road, 26, 'rgba(122,101,68,0.9)')
    72→
    73→  // Площадь деревни
    74→  ctx.fillStyle = 'rgba(110,90,60,0.85)'
    75→  ctx.beginPath()
    76→  ctx.ellipse(960, 1560, 190, 130, 0, 0, Math.PI * 2)
    77→  ctx.fill()
    78→  ctx.fillStyle = 'rgba(130,108,72,0.9)'
    79→  ctx.beginPath()
    80→  ctx.ellipse(960, 1560, 120, 84, 0, 0, Math.PI * 2)
    81→  ctx.fill()
    82→
    83→  // Пятно лагеря
    84→  ctx.fillStyle = 'rgba(96,78,52,0.8)'
    85→  ctx.beginPath()
    86→  ctx.ellipse(2590, 1650, 130, 95, 0, 0, Math.PI * 2)
    87→  ctx.fill()
    88→
    89→  // Круг Камней Древних
    90→  ctx.fillStyle = 'rgba(105,88,60,0.8)'
    91→  ctx.beginPath()
    92→  ctx.ellipse(2200, 660, 120, 95, 0, 0, Math.PI * 2)
    93→  ctx.fill()
    94→  ctx.strokeStyle = 'rgba(217,164,65,0.30)'
    95→  ctx.lineWidth = 3
    96→  ctx.beginPath()
    97→  ctx.ellipse(2200, 660, 92, 74, 0, 0, Math.PI * 2)
    98→  ctx.stroke()
    99→
   100→  // Озеро Лун
   101→  ctx.fillStyle = 'rgba(138,122,84,0.55)'
   102→  ctx.beginPath()
   103→  ctx.arc(LAKE.x, LAKE.y, LAKE.r + 30, 0, Math.PI * 2)
   104→  ctx.fill()
   105→  ctx.fillStyle = '#1d3d42'
   106→  ctx.beginPath()
   107→  ctx.arc(LAKE.x, LAKE.y, LAKE.r, 0, Math.PI * 2)
   108→  ctx.fill()
   109→  ctx.fillStyle = 'rgba(23,51,58,0.9)'
   110→  ctx.beginPath()
   111→  ctx.arc(LAKE.x, LAKE.y, LAKE.r - 34, 0, Math.PI * 2)
   112→  ctx.fill()
   113→  for (let i = 0; i < 12; i++) {
   114→    const a = rnd() * Math.PI * 2
   115→    const rr = 40 + rnd() * (LAKE.r - 60)
   116→    ctx.strokeStyle = 'rgba(190,220,210,0.14)'
   117→    ctx.lineWidth = 2
   118→    ctx.beginPath()
   119→    ctx.arc(LAKE.x + Math.cos(a) * rr, LAKE.y + Math.sin(a) * rr, 14 + rnd() * 20, 0, Math.PI)
   120→    ctx.stroke()
   121→  }
   122→
   123→  // Тёмная рамка мира
   124→  const edge = 90
   125→  const eg = ctx.createLinearGradient(0, 0, edge, 0)
   126→  eg.addColorStop(0, 'rgba(20,26,16,0.9)')
   127→  eg.addColorStop(1, 'rgba(20,26,16,0)')
   128→  ctx.fillStyle = eg
   129→  ctx.fillRect(0, 0, edge, WORLD.h)
   130→  ctx.save()
   131→  ctx.translate(WORLD.w, 0)
   132→  ctx.scale(-1, 1)
   133→  ctx.fillStyle = eg
   134→  ctx.fillRect(0, 0, edge, WORLD.h)
   135→  ctx.restore()
   136→  const eg2 = ctx.createLinearGradient(0, 0, 0, edge)
   137→  eg2.addColorStop(0, 'rgba(20,26,16,0.9)')
   138→  eg2.addColorStop(1, 'rgba(20,26,16,0)')
   139→  ctx.fillStyle = eg2
   140→  ctx.fillRect(0, 0, WORLD.w, edge)
   141→  ctx.save()
   142→  ctx.translate(0, WORLD.h)
   143→  ctx.scale(1, -1)
   144→  ctx.fillStyle = eg2
   145→  ctx.fillRect(0, 0, WORLD.w, edge)
   146→  ctx.restore()
   147→
   148→  return c
   149→}

   153→export function drawProp(ctx: CanvasRenderingContext2D, p: PropDef, t: number) {
   154→  const { pos } = p
   155→  switch (p.kind) {

   164→      const layers = [

   173→        ctx.lineTo(pos.x + L.w * s, pos.y + L.y * s)

   182→      ctx.lineTo(pos.x + 8 * s, pos.y - 74 * s)

   210→    }
   211→    case 'house': {

   213→      const h = p.h ?? 70

   448→      ctx.arc(pos.x, pos.y - 8, 3, 0, Math.PI * 2)
   449→      ctx.fill()
   450→      break
   451→    }
   452→  }
   453→}

   471→function drawWeaponSwing(ctx: CanvasRenderingContext2D, e: Entity, color: string) {
   472→  if (e.attackT <= 0) return
   473→  const p = 1 - e.attackT / 0.28
   474→  const a0 = e.facing - 1.1 + p * 2.2
   475→  ctx.save()
   476→  ctx.translate(e.pos.x, e.pos.y - 6)
   477→  ctx.rotate(a0)
   478→  ctx.strokeStyle = color
   479→  ctx.lineWidth = 3.5
   480→  ctx.lineCap = 'round'
   481→  ctx.beginPath()
   482→  ctx.moveTo(e.radius * 0.4, 0)
   483→  ctx.lineTo(e.radius * 1.5, 0)
   484→  ctx.stroke()
   485→  ctx.restore()
   486→  ctx.strokeStyle = `rgba(255,240,200,${0.35 * (1 - p)})`
   487→  ctx.lineWidth = 2
   488→  ctx.beginPath()
   489→  ctx.arc(e.pos.x, e.pos.y - 6, e.radius * 1.35, e.facing - 0.7, e.facing + 0.7)
   490→  ctx.stroke()
   491→}

   493→function drawHumanoid(
   494→  ctx: CanvasRenderingContext2D,
   495→  e: Entity,
   496→  t: number,
   497→  cloak: string,
   498→  weapon: 'sword' | 'staff' | 'shield' | 'bow' | 'none',
   499→) {
   500→  const bob = Math.sin(e.walkPhase * 10) * (Math.hypot(e.vel.x, e.vel.y) > 4 ? 1.6 : 0.4)
   501→  shadowBlob(ctx, e.pos.x, e.pos.y, e.radius)
   502→  const bx = e.pos.x
   503→  const by = e.pos.y - 8 + bob
   504→  // тело-плащ
   505→  ctx.fillStyle = e.hitT > 0 ? '#ffffff' : cloak
   506→  ctx.beginPath()
   507→  ctx.ellipse(bx, by, e.radius * 0.78, e.radius * 0.95, 0, 0, Math.PI * 2)
   508→  ctx.fill()
   509→  ctx.fillStyle = 'rgba(0,0,0,0.18)'
   510→  ctx.beginPath()
   511→  ctx.ellipse(bx - e.radius * 0.25, by + 3, e.radius * 0.4, e.radius * 0.7, 0, 0, Math.PI * 2)
   512→  ctx.fill()
   513→  // голова
   514→  ctx.fillStyle = '#d8b088'
   515→  ctx.beginPath()
   516→  ctx.arc(bx, by - e.radius * 0.95, e.radius * 0.5, 0, Math.PI * 2)
   517→  ctx.fill()
   518→  ctx.fillStyle = '#3a2c1c'
   519→  ctx.beginPath()
   520→  ctx.arc(bx, by - e.radius * 1.15, e.radius * 0.42, Math.PI, Math.PI * 2)
   521→  ctx.fill()
   522→  // оружие
   523→  const fx = Math.cos(e.facing)
   524→  const fy = Math.sin(e.facing)
   525→  if (weapon === 'shield') {
   526→    ctx.fillStyle = '#7a8a4a'
   527→    ctx.beginPath()
   528→    ctx.ellipse(bx + fx * e.radius * 0.9 - fy * e.radius * 0.5, by + fy * e.radius * 0.9 + fx * e.radius * 0.5, e.radius * 0.45, e.radius * 0.6, e.facing, 0, Math.PI * 2)
   529→    ctx.fill()
   530→    ctx.strokeStyle = '#4a522c'
   531→    ctx.lineWidth = 2
   532→    ctx.stroke()
   533→    drawWeaponSwing(ctx, e, '#c9c4bc')
   534→  } else if (weapon === 'staff') {
   535→    ctx.strokeStyle = '#6a5638'
   536→    ctx.lineWidth = 3
   537→    ctx.beginPath()
   538→    ctx.moveTo(bx - fy * e.radius * 0.7, by + fx * e.radius * 0.7)
   539→    ctx.lineTo(bx - fy * e.radius * 0.7 + fx * 6, by + fx * e.radius * 0.7 + fy * 6 - e.radius * 1.6)
   540→    ctx.stroke()
   541→    const glow = 0.6 + 0.4 * Math.sin(t * 4 + e.eid)
   542→    ctx.fillStyle = `rgba(159,184,164,${glow})`
   543→    ctx.beginPath()
   544→    ctx.arc(bx - fy * e.radius * 0.7 + fx * 6, by + fx * e.radius * 0.7 + fy * 6 - e.radius * 1.7, 3.5, 0, Math.PI * 2)
   545→    ctx.fill()
   546→  } else if (weapon === 'bow') {
   547→    ctx.strokeStyle = '#8a7a54'
   548→    ctx.lineWidth = 2.5
   549→    ctx.beginPath()
   550→    ctx.arc(bx + fx * e.radius, by + fy * e.radius, e.radius * 0.8, e.facing - 1.2, e.facing + 1.2)
   551→    ctx.stroke()
   552→  } else if (weapon === 'sword') {
   553→    drawWeaponSwing(ctx, e, '#c9c4bc')
   554→  }
   555→  // щит Квена
   556→  if (e.shield > 0) {
   557→    ctx.strokeStyle = `rgba(217,164,65,${0.5 + 0.3 * Math.sin(t * 6)})`
   558→    ctx.lineWidth = 2.5
   559→    ctx.beginPath()
   560→    ctx.arc(bx, by - 4, e.radius + 7, 0, Math.PI * 2)
   561→    ctx.stroke()
   562→  }
   563→}

   565→function drawWolf(ctx: CanvasRenderingContext2D, e: Entity, big: boolean) {
   566→  const s = big ? 1.45 : 1
   567→  const bob = Math.sin(e.walkPhase * 12) * (Math.hypot(e.vel.x, e.vel.y) > 4 ? 1.4 : 0.3)
   568→  shadowBlob(ctx, e.pos.x, e.pos.y, e.radius * s)
   569→  const bx = e.pos.x
   570→  const by = e.pos.y - 7 * s + bob
   571→  ctx.save()
   572→  ctx.translate(bx, by)
   573→  ctx.rotate(e.facing)
   574→  ctx.fillStyle = e.hitT > 0 ? '#ffffff' : big ? '#7d7466' : '#565a5c'
   575→  ctx.beginPath()
   576→  ctx.ellipse(0, 0, e.radius * 1.25 * s, e.radius * 0.72 * s, 0, 0, Math.PI * 2)
   577→  ctx.fill()
   578→  // голова
   579→  ctx.beginPath()
   580→  ctx.ellipse(e.radius * 1.05 * s, -2 * s, e.radius * 0.55 * s, e.radius * 0.45 * s, 0, 0, Math.PI * 2)
   581→  ctx.fill()
   582→  // уши
   583→  ctx.fillStyle = '#3d3a36'
   584→  ctx.beginPath()
   585→  ctx.moveTo(e.radius * 0.9 * s, -e.radius * 0.5 * s)
   586→  ctx.lineTo(e.radius * 1.1 * s, -e.radius * 1.0 * s)
   587→  ctx.lineTo(e.radius * 1.25 * s, -e.radius * 0.4 * s)
   588→  ctx.closePath()
   589→  ctx.fill()
   590→  // хвост
   591→  ctx.strokeStyle = e.hitT > 0 ? '#ffffff' : big ? '#6d6455' : '#484c4e'
   592→  ctx.lineWidth = 4 * s
   593→  ctx.beginPath()
   594→  ctx.moveTo(-e.radius * 1.1 * s, 0)
   595→  ctx.quadraticCurveTo(-e.radius * 1.7 * s, Math.sin(e.walkPhase * 9) * 4, -e.radius * 2 * s, -4 * s)
   596→  ctx.stroke()
   597→  // глаза
   598→  ctx.fillStyle = big ? '#e0653a' : '#d9a441'
   599→  ctx.beginPath()
   600→  ctx.arc(e.radius * 1.35 * s, -4 * s, 1.8 * s, 0, Math.PI * 2)
   601→  ctx.arc(e.radius * 1.35 * s, 1 * s, 1.8 * s, 0, Math.PI * 2)
   602→  ctx.fill()
   603→  ctx.restore()
   604→}

   606→function drawHorror(ctx: CanvasRenderingContext2D, e: Entity, t: number) {
   607→  shadowBlob(ctx, e.pos.x, e.pos.y, e.radius)
   608→  const by = e.pos.y - 12 + Math.sin(t * 3 + e.eid) * 2
   609→  ctx.fillStyle = e.hitT > 0 ? '#ffffff' : 'rgba(122,90,146,0.85)'
   610→  ctx.beginPath()
   611→  ctx.arc(e.pos.x, by, e.radius * 0.85, 0, Math.PI * 2)
   612→  ctx.fill()
   613→  ctx.fillStyle = 'rgba(40,25,55,0.9)'
   614→  ctx.beginPath()
   615→  ctx.arc(e.pos.x + 3, by - 2, e.radius * 0.4, 0, Math.PI * 2)
   616→  ctx.fill()
   617→  ctx.fillStyle = 'rgba(220,170,255,0.9)'
   618→  ctx.beginPath()
   619→  ctx.arc(e.pos.x + 4, by - 3, 2.2, 0, Math.PI * 2)
   620→  ctx.fill()
   621→  ctx.strokeStyle = 'rgba(122,90,146,0.7)'
   622→  ctx.lineWidth = 2.5
   623→  for (let i = 0; i < 4; i++) {
   624→    const a = (i / 4) * Math.PI * 2 + t * 1.5
   625→    ctx.beginPath()
   626→    ctx.moveTo(e.pos.x, by)
   627→    ctx.quadraticCurveTo(
   628→      e.pos.x + Math.cos(a) * e.radius * 1.2, by + Math.sin(a) * e.radius * 0.8 + 4,
   629→      e.pos.x + Math.cos(a) * e.radius * 1.5, e.pos.y + Math.sin(a) * e.radius * 0.5 + 8,
   630→    )
   631→    ctx.stroke()
   632→  }
   633→}

   635→function drawBoss(ctx: CanvasRenderingContext2D, e: Entity, t: number) {
   636→  shadowBlob(ctx, e.pos.x, e.pos.y, e.radius * 1.2)
   637→  const by = e.pos.y - 16 + Math.sin(t * 2) * 2
   638→  const pulse = 0.5 + 0.5 * Math.sin(t * 5)
   639→  // аура
   640→  const g = ctx.createRadialGradient(e.pos.x, by, 6, e.pos.x, by, e.radius * 2.4)
   641→  g.addColorStop(0, `rgba(160,107,217,${0.25 + pulse * 0.12})`)
   642→  g.addColorStop(1, 'rgba(160,107,217,0)')
   643→  ctx.fillStyle = g
   644→  ctx.fillRect(e.pos.x - e.radius * 2.4, by - e.radius * 2.4, e.radius * 4.8, e.radius * 4.8)
   645→  // масса
   646→  ctx.fillStyle = e.hitT > 0 ? '#ffffff' : '#3d2f4d'
   647→  ctx.beginPath()
   648→  ctx.ellipse(e.pos.x, by, e.radius * 1.05, e.radius * 1.25, 0, 0, Math.PI * 2)
   649→  ctx.fill()
   650→  // шипы-рога
   651→  ctx.fillStyle = '#2a1f38'
   652→  for (let i = -2; i <= 2; i++) {
   653→    if (i === 0) continue
   654→    ctx.beginPath()
   655→    ctx.moveTo(e.pos.x + i * e.radius * 0.38, by - e.radius * 0.9)
   656→    ctx.lineTo(e.pos.x + i * e.radius * 0.6, by - e.radius * 2.0)
   657→    ctx.lineTo(e.pos.x + i * e.radius * 0.75, by - e.radius * 0.7)
   658→    ctx.closePath()
   659→    ctx.fill()
   660→  }
   661→  // ядро
   662→  ctx.fillStyle = `rgba(208,138,58,${0.75 + pulse * 0.25})`
   663→  ctx.beginPath()
   664→  ctx.arc(e.pos.x, by, e.radius * 0.42 + pulse * 2, 0, Math.PI * 2)
   665→  ctx.fill()
   666→  // глаза
   667→  ctx.fillStyle = '#d97a3a'
   668→  ctx.beginPath()
   669→  ctx.arc(e.pos.x - 8, by - e.radius * 0.5, 3, 0, Math.PI * 2)
   670→  ctx.arc(e.pos.x + 8, by - e.radius * 0.5, 3, 0, Math.PI * 2)
   671→  ctx.fill()
   672→}

   674→export function drawEntity(
   675→  ctx: CanvasRenderingContext2D,
   676→  e: Entity,
   677→  t: number,
   678→  isTarget: boolean,
   679→) {
   680→  if (!e.alive) return
   681→  if (isTarget) {
   682→    ctx.strokeStyle = 'rgba(217,164,65,0.85)'
   683→    ctx.lineWidth = 2
   684→    ctx.setLineDash([6, 5])
   685→    ctx.beginPath()
   686→    ctx.ellipse(e.pos.x, e.pos.y + e.radius * 0.55, e.radius + 9, (e.radius + 9) * 0.45, 0, 0, Math.PI * 2)
   687→    ctx.stroke()
   688→    ctx.setLineDash([])
   689→  }
   690→  switch (e.enemyType) {
   691→    case 'wolf':
   692→      drawWolf(ctx, e, false)
   693→      break
   694→    case 'alpha':
   695→      drawWolf(ctx, e, true)
   696→      break
   697→    case 'horror':
   698→      drawHorror(ctx, e, t)
   699→      break
   700→    case 'boss':
   701→      drawBoss(ctx, e, t)
   702→      break
   703→    default: {
   704→      const weapon =
   705→        e.id === 'kaya'
   706→          ? 'staff'
   707→          : e.id === 'torvald'
   708→            ? 'shield'
   709→            : e.enemyType === 'archer'
   710→              ? 'bow'
   711→              : e.enemyType
   712→                ? 'sword'
   713→                : e.role === 'Архимаг Ковена' || e.role === 'Магистр-странница'
   714→                  ? 'staff'
   715→                  : 'sword'
   716→      drawHumanoid(ctx, e, t, e.color, weapon)
   717→    }
   718→  }
   719→  // оглушение
   720→  if (e.stunT > 0) {
   721→    for (let i = 0; i < 3; i++) {
   722→      const a = t * 5 + (i / 3) * Math.PI * 2
   723→      ctx.fillStyle = '#f5c96b'
   724→      ctx.beginPath()
   725→      ctx.arc(e.pos.x + Math.cos(a) * 10, e.pos.y - e.radius * 2.4 + Math.sin(a) * 4, 2.2, 0, Math.PI * 2)
   726→      ctx.fill()
   727→    }
   728→  }
   729→  // горение
   730→  if (e.burnT > 0) {
   731→    for (let i = 0; i < 2; i++) {
   732→      const px = e.pos.x + Math.sin(t * 12 + i * 2 + e.eid) * 6
   733→      ctx.fillStyle = `rgba(224,138,58,${0.5 + Math.random() * 0.3})`
   734→      ctx.beginPath()
   735→      ctx.arc(px, e.pos.y - 6 - Math.random() * 14, 2.4, 0, Math.PI * 2)
   736→      ctx.fill()
   737→    }
   738→  }
   739→  // замедление
   740→  if (e.slowT > 0) {
   741→    ctx.strokeStyle = 'rgba(200,230,240,0.5)'
   742→    ctx.lineWidth = 1.5
   743→    ctx.beginPath()
   744→    ctx.arc(e.pos.x, e.pos.y, e.radius + 4, 0, Math.PI * 2)
   745→    ctx.stroke()
   746→  }
   747→  // каст
   748→  if (e.castT > 0) {
   749→    ctx.strokeStyle = 'rgba(159,184,164,0.6)'
   750→    ctx.lineWidth = 2
   751→    ctx.beginPath()
   752→    ctx.arc(e.pos.x, e.pos.y, e.radius + 10 - (0.4 - e.castT) * 20, 0, Math.PI * 2)
   753→    ctx.stroke()
   754→  }
   755→  // полоса HP
   756→  if (e.kind === 'enemy' && e.hp < e.maxHp) {
   757→    hpBar(ctx, e.pos.x, e.pos.y - e.radius * 2.2 - 8, Math.max(26, e.radius * 2), e.hp / e.maxHp, e.boss ? '#a06bd9' : '#c94f3d')
   758→  }
   759→  if (e.kind === 'companion' && e.hp < e.maxHp) {
   760→    hpBar(ctx, e.pos.x, e.pos.y - e.radius * 2.2 - 8, 26, e.hp / e.maxHp, '#8fbf7f')
   761→  }
   762→  // имя
   763→  if (e.kind === 'npc' || e.kind === 'companion') {
   764→    ctx.font = '11px ui-sans-serif, system-ui'
   765→    ctx.textAlign = 'center'
   766→    ctx.fillStyle = 'rgba(0,0,0,0.55)'
   767→    const tw = ctx.measureText(e.name).width
   768→    ctx.fillRect(e.pos.x - tw / 2 - 4, e.pos.y - e.radius * 2.2 - 24, tw + 8, 14)
   769→    ctx.fillStyle = e.kind === 'companion' ? '#c9e0b8' : '#e8d5a8'
   770→    ctx.fillText(e.name, e.pos.x, e.pos.y - e.radius * 2.2 - 13)
   771→  }
   772→}

   813→export function drawMinimap(
   814→  mini: HTMLCanvasElement,
   815→  baked: HTMLCanvasElement,
   816→  entities: Entity[],
   817→  playerPos: Vec,
   818→  questMarker: Vec | null,
   819→  size: number,
   820→  t: number,
   821→) {
   822→  const ctx = mini.getContext('2d')
   823→  if (!ctx) return
   824→  const scale = size / WORLD.w
   825→  ctx.clearRect(0, 0, size, size)
   826→  ctx.drawImage(baked, 0, 0, WORLD.w, WORLD.h, 0, 0, size, size * (WORLD.h / WORLD.w))
   827→  const mh = size * (WORLD.h / WORLD.w)
   828→  const dot = (x: number, y: number, color: string, r: number) => {
   829→    ctx.fillStyle = color
   830→    ctx.beginPath()
   831→    ctx.arc(x * scale, y * (mh / WORLD.h), r, 0, Math.PI * 2)
   832→    ctx.fill()
   833→  }
   834→  for (const e of entities) {
   835→    if (!e.alive) continue
   836→    if (e.kind === 'npc') dot(e.pos.x, e.pos.y, '#e8d5a8', 2)
   837→    else if (e.kind === 'companion') dot(e.pos.x, e.pos.y, '#c9e0b8', 2)
   838→    else if (e.kind === 'enemy' && e.aiState !== 'return') dot(e.pos.x, e.pos.y, '#c94f3d', 2)
   839→  }
   840→  if (questMarker) {
   841→    const pulse = 2 + Math.sin(t * 4) * 1
   842→    ctx.strokeStyle = '#f5c96b'
   843→    ctx.lineWidth = 1.5
   844→    ctx.beginPath()
   845→    ctx.arc(questMarker.x * scale, questMarker.y * (mh / WORLD.h), 4 + pulse, 0, Math.PI * 2)
   846→    ctx.stroke()
   847→  }
   848→  // игрок
   849→  ctx.fillStyle = '#f5c96b'
   850→  ctx.beginPath()
   851→  ctx.arc(playerPos.x * scale, playerPos.y * (mh / WORLD.h), 3, 0, Math.PI * 2)
   852→  ctx.fill()
   853→  ctx.strokeStyle = 'rgba(255,255,255,0.7)'
   854→  ctx.lineWidth = 1
   855→  ctx.stroke()
   856→}

   858→export function drawFullMap(
   859→  canvas: HTMLCanvasElement,
   860→  baked: HTMLCanvasElement,
   861→  playerPos: Vec,
   862→  questMarker: Vec | null,
   863→) {
   864→  const ctx = canvas.getContext('2d')
   865→  if (!ctx) return
   866→  const w = canvas.width
   867→  const h = canvas.height
   868→  ctx.clearRect(0, 0, w, h)
   869→  ctx.drawImage(baked, 0, 0, WORLD.w, WORLD.h, 0, 0, w, h)
   870→  const sx = w / WORLD.w
   871→  const sy = h / WORLD.h
   872→  ctx.font = '12px ui-serif, Georgia, serif'
   873→  ctx.textAlign = 'center'
   874→  for (const m of AREA_MARKERS) {
   875→    ctx.fillStyle = 'rgba(20,16,12,0.6)'
   876→    const tw = ctx.measureText(m.label).width
   877→    ctx.fillRect(m.pos.x * sx - tw / 2 - 5, m.pos.y * sy - 18, tw + 10, 16)
   878→    ctx.fillStyle = '#e8d5a8'
   879→    ctx.fillText(m.label, m.pos.x * sx, m.pos.y * sy - 6)
