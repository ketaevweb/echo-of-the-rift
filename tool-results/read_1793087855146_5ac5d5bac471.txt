    38→export function dialogueCtx(ctx: EngineCtx) {
    39→  return {
    40→    quest: (id: string) => ctx.quests[id],
    41→    flag: (name: string) => !!ctx.flags[name],
    42→    faction: ctx.faction,
    43→    companion: (id: string) => ctx.companionIds[id as 'kaya' | 'torvald'],
    44→    has: (item: string, qty = 1) => countItem(ctx, item) >= qty,
    45→    lp: ctx.lp,
    46→    gold: ctx.gold,
    47→  }
    48→}

    50→export function evalCond(ctx: EngineCtx, cond: string): boolean {
    51→  if (!cond) return true
    52→  for (const tok of cond.split(',')) {
    53→    const s = tok.trim()
    54→    if (s && !evalToken(ctx, s)) return false
    55→  }
    56→  return true
    57→}
    58→
    59→function evalToken(ctx: EngineCtx, s: string): boolean {
    60→  let m = s.match(/^quest\.(\w+)==(\d+)$/)
    61→  if (m) return ctx.quests[m[1]].stage === parseInt(m[2], 10)
    62→  m = s.match(/^quest\.(\w+)>=(\d+)$/)
    63→  if (m) return ctx.quests[m[1]].stage >= parseInt(m[2], 10)
    64→  m = s.match(/^quest\.(\w+)<(\d+)$/)
    65→  if (m) return ctx.quests[m[1]].stage < parseInt(m[2], 10)
    66→  m = s.match(/^flag\.(\w+)$/)
    67→  if (m) return !!ctx.flags[m[1]]
    68→  m = s.match(/^!flag\.(\w+)$/)
    69→  if (m) return !ctx.flags[m[1]]
    70→  m = s.match(/^faction==(\w+)$/)
    71→  if (m) return ctx.faction === (m[1] as Faction)
    72→  m = s.match(/^comp\.(\w+)$/)
    73→  if (m) return !!ctx.companionIds[m[1] as 'kaya' | 'torvald']
    74→  m = s.match(/^!comp\.(\w+)$/)
    75→  if (m) return !ctx.companionIds[m[1] as 'kaya' | 'torvald']
    76→  m = s.match(/^has\.(\w+)(?::(\d+))?$/)
    77→  if (m) return countItem(ctx, m[1]) >= (m[2] ? parseInt(m[2], 10) : 1)
    78→  m = s.match(/^lp>=(\d+)$/)
    79→  if (m) return ctx.lp >= parseInt(m[1], 10)
    80→  m = s.match(/^gold>=(\d+)$/)
    81→  if (m) return ctx.gold >= parseInt(m[1], 10)
    82→  m = s.match(/^origin:(\w+)$/)
    83→  if (m) return ctx.origin === (m[1] as OriginId)
    84→  // Фаза 6: карьера Гранита
    85→  m = s.match(/^career==([\w-]+)$/)
    86→  if (m) return ctx.career.faction === (m[1] as CareerId)
    87→  m = s.match(/^!career$/)
    88→  if (m) return ctx.career.faction === null
    89→  m = s.match(/^career$/)
    90→  if (m) return ctx.career.faction !== null
    91→  m = s.match(/^rank>=(\d+)$/)
    92→  if (m) return ctx.career.faction !== null && ctx.career.rank[ctx.career.faction] >= parseInt(m[1], 10)
    93→  m = s.match(/^rep:([\w-]+)>=(\d+)$/)
    94→  if (m) return ctx.career.rep[m[1] as CareerId] >= parseInt(m[2], 10)
    95→  return true
    96→}

    98→export function openDialogue(ctx: EngineCtx, treeId: string) {
    99→  const tree = DIALOGUES[treeId]
   100→  if (!tree) return
   101→  ctx.dialogueTree = treeId
   102→  const start = typeof tree.start === 'function' ? tree.start(dialogueCtx(ctx)) : tree.start
   103→  setNode(ctx, start)
   104→}
   105→
   106→export function setNode(ctx: EngineCtx, nodeId: string) {
   107→  const tree = DIALOGUES[ctx.dialogueTree!]
   108→  const node = tree.nodes[nodeId]
   109→  if (!node) {
   110→    closeDialogue(ctx)
   111→    return
   112→  }
   113→  const choices = node.choices.filter((c) => evalCond(ctx, c.cond ?? ''))
   114→  ctx.dlgChoices = choices
   115→  const p = PORTRAITS[node.speaker] ?? { icon: 'User', color: '#b08356', title: node.role ?? '' }
   116→  // Фаза 5: кинокамера — ищем сущность по имени спикера рядом с героем
   117→  ctx.dialogueFocusEid = findSpeakerEntity(ctx, node.speaker)
   118→  useGame.getState().patch({
   119→    dialogue: {
   120→      speaker: node.speaker,
   121→      role: p.title,
   122→      portraitIcon: p.icon,
   123→      portraitColor: p.color,
   124→      text: node.text,
   125→      choices: choices.map((c) => ({
   126→        text: c.text,
   127→        tag: c.tag,
   128→        checkLabel: c.check ? c.check.label : undefined,
   129→        origin: !!c.origin,
   130→      })),
   131→      timer: node.timer,
   132→      onTimeout: node.onTimeout,
   133→    },
   134→    diceRoll: null, // новый узел — старый бросок не должен висеть
   135→  })
   136→}

   138→/** Сущность спикера для кинокамеры: NPC/уникальный враг с тем же именем в 450 ед. от героя. */
   139→function findSpeakerEntity(ctx: EngineCtx, speaker: string): number | null {
   140→  let best: Entity | null = null
   141→  let bd = 450
   142→  for (const e of ctx.entities) {
   143→    if (e.kind !== 'npc' && e.kind !== 'enemy') continue
   144→    if (e.name !== speaker) continue
   145→    const d = Math.hypot(e.pos.x - ctx.player.pos.x, e.pos.y - ctx.player.pos.y)
   146→    if (d < bd) {
   147→      bd = d
   148→      best = e
   149→    }
   150→  }
   151→  return best ? best.eid : null
   152→}

   154→export function chooseOption(ctx: EngineCtx, i: number) {
   155→  const c = ctx.dlgChoices[i]
   156→  if (!c) return
   157→  // Фаза 5: пока кубик летит — повторный выбор игнорируем
   158→  if (useGame.getState().diceRoll) return
   159→  sfx('click')
   160→  if (c.check) {
   161→    const bonus = 1 + Math.floor(ctx.attributes.charisma / 2) + ctx.skills.speech
   162→    const roll = d(20)
   163→    const total = roll + bonus
   164→    const success = roll !== 1 && (roll === 20 || total >= c.check.dc)
   165→    ctx.toast(
   166→      `${c.check.label}: d20 ${roll} + ${bonus} = ${total} — ${success ? 'успех!' : 'провал'}`,
   167→      success ? 'quest' : 'warn',
   168→    )
   169→    // Фаза 5: видимый бросок (BG3) — оверлей, потом продолжение ветки
   170→    useGame.getState().patch({ diceRoll: { label: c.check.label, roll, bonus, total, dc: c.check.dc, success } })
   171→    sfx(success ? 'quest' : 'warn')
   172→    window.setTimeout(() => {
   173→      if (ctx.destroyed) return
   174→      useGame.getState().patch({ diceRoll: null })
   175→      if (success && c.action) runAction(ctx, c.action)
   176→      const next = success ? c.goto : (c.gotoFail ?? null)
   177→      if (next === null || next === undefined) {
   178→        closeDialogue(ctx)
   179→        return
   180→      }
   181→      setNode(ctx, next)
   182→    }, 1700)
   183→    return
   184→  }
   185→  if (c.action) runAction(ctx, c.action)
   186→  if (c.goto === null || c.goto === undefined) {
   187→    closeDialogue(ctx)
   188→    return
   189→  }
   190→  setNode(ctx, c.goto)
   191→}

   193→export function closeDialogue(ctx: EngineCtx) {
   194→  ctx.dialogueTree = null
   195→  ctx.dlgChoices = []
   196→  ctx.dialogueFocusEid = null // кинокамера возвращается к герою
   197→  useGame.getState().patch({ dialogue: null, diceRoll: null })
   198→}

   200→export function runAction(ctx: EngineCtx, action: string) {
   201→  for (const cmdRaw of action.split(';')) {
   202→    const cmd = cmdRaw.trim()
   203→    if (!cmd || cmd === 'end') continue
   204→    const [op, arg1, arg2] = cmd.split(':')
   205→    switch (op) {
   206→      case 'quest':
   207→        advanceQuest(ctx, arg1, parseInt(arg2, 10))
   208→        break
   209→      case 'flag':
   210→        ctx.flags[arg1] = true
   211→        break
   212→      case 'give':
   213→        addItem(ctx, arg1, parseInt(arg2 ?? '1', 10))
   214→        break
   215→      case 'take':
   216→        removeItem(ctx, arg1, parseInt(arg2 ?? '1', 10))
   217→        break
   218→      case 'gold': {
   219→        ctx.gold = Math.max(0, ctx.gold + parseInt(arg1, 10))
   220→        sfx('coin')
   221→        break
   222→      }
   223→      case 'xp':
   224→        gainXp(ctx, parseInt(arg1, 10))
   225→        break
   226→      case 'silence':
   227→        // Фаза 5: Тишь — дешёвые решения +=N, человечные -=N; мир выцветает
   228→        ctx.silence = Math.max(0, Math.min(100, ctx.silence + parseInt(arg1, 10)))
   229→        break
   230→      case 'lp':
   231→        ctx.lp = Math.max(0, ctx.lp + parseInt(arg1, 10))
   232→        break
   233→      case 'open':
   234→        closeDialogue(ctx)
   235→        if (arg1 === 'shop') openShop(ctx, arg2 === 'alra' ? 'alra' : arg2 === 'brock' ? 'brock' : 'grim')
   236→        if (arg1 === 'trainer') openTrainer(ctx)
   237→        break
   238→      case 'bestiary':
   239→        if (!ctx.bestiary.has(arg1)) {
   240→          ctx.bestiary.add(arg1)
   241→          ctx.toast('Бестиарий: ' + (BESTIARY_INFO[arg1]?.name ?? arg1), 'quest')
   242→          sfx('quest')
   243→        }
   244→        break
   245→      case 'join':
   246→        if (arg1 === 'kaya' || arg1 === 'torvald') {
   247→          const wasNew = !ctx.companionIds[arg1]
   248→          ctx.spawnCompanion(arg1)
   249→          if (wasNew) {
   250→            ctx.toast(`${arg1 === 'kaya' ? 'Кайя' : 'Торвальд'} присоединяется к отряду!`, 'quest')
   251→            sfx('quest')
   252→          }
   253→        }
   254→        break
   255→      case 'faction':
   256→        ctx.faction = arg1 as Faction
   257→        ctx.toast(arg1 === 'flame' ? 'Ты вступил в Орден Пламени!' : 'Ты вступил в Ковен Тумана!', 'quest')
   258→        if (arg1 === 'mist') checkProof(ctx)
   259→        sfx('quest')
   260→        break
   261→      // Фаза 6: карьера Гранита — вербовка/ранг/слава, ямные бои
   262→      case 'career':
   263→        if (arg1 === 'join') joinCareer(ctx, arg2 as CareerId)
   264→        if (arg1 === 'rank') rankUp(ctx, arg2 as CareerId)
   265→        break
   266→      case 'rep':
   267→        addRep(ctx, arg1 as CareerId, parseInt(arg2, 10))
   268→        break
   269→      case 'pit':
   270→        if (arg1 === 'start') {
   271→          closeDialogue(ctx)
   272→          startPitFight(ctx, parseInt(arg2, 10))
   273→        }
   274→        break
   275→      case 'key':
   276→        // Ключ-осколок домена (Фаза 6): флаг + предмет-трофей
   277→        ctx.flags[arg1] = true
   278→        addItem(ctx, arg1 === 'key_granite' ? 'core_key' : arg1, 1)
   279→        break
   280→      case 'sign':
   281→        if (arg1 === 'shout' && !ctx.signs.shout.unlocked) {
   282→          ctx.signs.shout.unlocked = true
   283→          ctx.toast('Голос Древних изучен! Клавиша 5.', 'level')
   284→          sfx('level')
   285→        }
   286→        break
   287→      case 'boss':
   288→        if (arg1 === 'fight') {
   289→          const boss = ctx.entities.find((e) => e.enemyType === 'boss' && e.alive)
   290→          if (boss) boss.aiState = 'chase'
   291→        } else if (arg1 === 'fight_pit') {
   292→          // Фаза 6: вызвать Мастера Ямы
   293→          const pm = ctx.entities.find((e) => e.enemyType === 'pit_master' && e.alive)
   294→          if (pm) {
   295→            pm.aggroR = 420
   296→            pm.aiState = 'chase'
   297→            floater(ctx, pm.pos, 'ЯМА ПРИНЯЛА ВЫЗОВ!', '#c9a44a', 16)
   298→            sfx('boss')
   299→          }
   300→        } else if (arg1 === 'weaken_pit') {
   301→          const pm = ctx.entities.find((e) => e.enemyType === 'pit_master' && e.alive)
   302→          if (pm) {
   303→            pm.hp = Math.round(pm.hp * 0.85)
   304→            pm.dmgMod = Math.max(1, pm.dmgMod - 2)
   305→            floater(ctx, pm.pos, 'Толпа усмехается Мастеру...', '#c9a44a', 15)
   306→          }
   307→        } else if (arg1 === 'weaken') {
   308→          const boss = ctx.entities.find((e) => e.enemyType === 'boss' && e.alive)
   309→          if (boss) {
   310→            boss.hp = Math.round(boss.hp * 0.8)
   311→            boss.dmgMod = 2
   312→            boss.summonStage = 1
   313→            floater(ctx, boss.pos, 'Слова жгут его!', '#c9a3e8', 15)
   314→          }
   315→        }
   316→        break
   317→      case 'heal':
   318→        if (arg1 === 'full') {
   319→          ctx.hp = ctx.maxHp
   320→          ctx.player.hp = ctx.hp
   321→        }
   322→        break
   323→      // Фаза 6.3: оборона Бастиона — бить рог
   324→      case 'siege':
   325→        if (arg1 === 'start') {
   326→          closeDialogue(ctx)
   327→          startSiege(ctx)
   328→        }
   329→        break
   330→    }
   331→  }
   332→  ctx.syncUi()
   333→}

   337→export function runInteractable(ctx: EngineCtx, it: InteractableDef) {
   338→  if (it.action.startsWith('dialogue:')) {
   339→    openDialogue(ctx, it.action.slice(9))
   340→  } else if (it.action.startsWith('portal:')) {
   341→    // Врата Тумана / Врата Гранита / Врата Пределья / Врата Мары (Фаза 3/6/6.2/6.3)
   342→    const zone = it.action.slice(7)
   343→    if (zone === 'veld' || zone === 'valley' || zone === 'granite' || zone === 'frost' || zone === 'mara') {
   344→      ctx.travelTo(zone as 'veld' | 'valley' | 'granite' | 'frost' | 'mara')
   345→    }
   346→  } else if (it.action.startsWith('purge:')) {
   347→    // Мор-сгустки Мары (Фаза 6.3): выжечь скверну в жилах — квест Хольта
   348→    const questId = it.action.slice(6)
   349→    ctx.removedProps.add(it.id)
   350→    ctx.flags['well_' + it.id] = true
   351→    ring(ctx, it.pos, '#f5c96b', 70, 0.9)
   352→    floater(ctx, it.pos, 'СКВЕРНА ВЫЖЕЧЕНА', '#f5c96b', 15)
   353→    sfx('fire')
   354→    const q = ctx.quests[questId]
   355→    if (q && !q.done) {
   356→      q.counter++
   357→      if (q.stage === 1 && q.counter >= 3) {
   358→        advanceQuest(ctx, questId, 2)
   359→      } else {
   360→        ctx.toast(`Мор-сгусток: ${q.counter}/3`, 'quest')
   361→        saveGame(ctx, false)
   362→        ctx.syncUi()
   363→      }
   364→    }
   365→  } else if (it.action.startsWith('rune:')) {
   366→    // Пещера Рун (Фаза 6.2): поворот рунного камня-кольца
   367→    cycleRune(ctx, it.action.slice(5))
   368→  } else if (it.action.startsWith('word:')) {
   369→    // Стена Слова (Фаза 6.2): взять слово силы Ту'ум
   370→    learnWord(ctx, it.action.slice(5) === 'frost' ? 'frost' : 'force')
   371→  } else if (it.action.startsWith('fclue:')) {
   372→    // Следы пожара Стылого Пределья (Фаза 6.2): цепочка к Драконьему Долу
   373→    const questId = it.action.slice(6)
   374→    ctx.removedProps.add(it.id)
   375→    ctx.flags['clue_' + it.id] = true
   376→    const clue = FROST_CLUES.find((c) => c.id === it.id)
   377→    floater(ctx, it.pos, clue?.name ?? 'След', '#bfe3ea', 14)
   378→    if (clue) ctx.toast(clue.text, 'info')
   379→    sfx('quest')
   380→    const q = ctx.quests[questId]
   381→    if (q && !q.done) {
   382→      q.counter++
   383→      if (q.stage === 1 && q.counter >= 3) {
   384→        advanceQuest(ctx, questId, 2)
   385→      } else {
   386→        ctx.toast(`След пожара: ${q.counter}/3`, 'quest')
   387→        saveGame(ctx, false)
   388→        ctx.syncUi()
   389→      }
   390→    }
   391→  } else if (it.action.startsWith('ore:')) {
   392→    // Гранитные жилы (Фаза 6): добыча руды для кузнеца
   393→    const item = it.action.slice(4)
   394→    addItem(ctx, item, 1)
   395→    ctx.removedProps.add(it.id)
   396→    ctx.flags['ore_' + it.id] = true
   397→    sfx('coin')
   398→    ctx.toast('Добыто: ' + ITEMS[item].name, 'loot')
   399→    const q = ctx.quests.granite_ore
   400→    if (q.stage === 1 && !q.done) {
   401→      q.counter++
   402→      if (q.counter >= 4) ctx.toast('Руды достаточно — к Броку в Колонию!', 'quest')
   403→      else ctx.toast(`Гранитная руда: ${q.counter}/4`, 'info')
   404→    }
   405→    ctx.syncUi()
   406→  } else if (it.action.startsWith('clue:')) {
   407→    // Следопытство W3 (Фаза 3): цепочка следов ведёт к логову
   408→    const questId = it.action.slice(5)
   409→    ctx.removedProps.add(it.id)
   410→    ctx.flags['clue_' + it.id] = true
   411→    const clue = VELD_CLUES.find((c) => c.id === it.id)
   412→    floater(ctx, it.pos, clue?.name ?? 'След', '#d9c04f', 14)
   413→    if (clue) ctx.toast(clue.text, 'info')
   414→    sfx('quest')
   415→    const q = ctx.quests[questId]
   416→    if (q && !q.done) {
   417→      q.counter++
   418→      if (q.stage === 1 && q.counter >= 3) {
   419→        ctx.flags.ashMotherRevealed = true
   420→        spawnAshMother(ctx)
   421→        advanceQuest(ctx, questId, 2)
   422→      } else {
   423→        ctx.toast(`След: ${q.counter}/3`, 'quest')
   424→        saveGame(ctx, false)
   425→        ctx.syncUi()
   426→      }
   427→    }
   428→  } else if (it.action.startsWith('gather:')) {
   429→    const item = it.action.slice(7)
   430→    addItem(ctx, item, 1)
   431→    ctx.removedProps.add(it.id)
   432→    ctx.flags['herb_' + it.id] = true
   433→    sfx('coin')
   434→    ctx.toast('Собрано: ' + ITEMS[item].name, 'loot')
   435→    const q = ctx.quests.mist
   436→    if (q.stage === 1 && !q.done) {
   437→      q.counter++
   438→      if (q.counter >= 3) ctx.toast('Лунные травы собраны — к Селене!', 'quest')
   439→      else ctx.toast(`Лунная трава: ${q.counter}/3`, 'info')
   440→    }
   441→    ctx.syncUi()
   442→  }
   443→}

   445→export function pickInteractable(ctx: EngineCtx, p: Vec, extra = 0): InteractableDef | null {
   446→  let best: InteractableDef | null = null
   447→  let bd = 1e9
   448→  for (const it of INTERACTABLES) {
   449→    if (ctx.removedProps.has(it.id)) continue
   450→    if (!evalCond(ctx, it.cond ?? '')) continue
   451→    const dist = Math.hypot(it.pos.x - p.x, it.pos.y - p.y)
   452→    if (dist < it.r + extra && dist < bd) {
   453→      bd = dist
   454→      best = it
   455→    }
   456→  }
   457→  return best
   458→}

   460→export function nearestInteraction(ctx: EngineCtx): { label: string; npc?: Entity; it?: InteractableDef } | null {
   461→  let best: { label: string; npc?: Entity; it?: InteractableDef } | null = null
   462→  let bd = 1e9
   463→  for (const e of ctx.entities) {
   464→    if (e.kind !== 'npc' || !e.alive) continue
   465→    const dist = Math.hypot(e.pos.x - ctx.player.pos.x, e.pos.y - ctx.player.pos.y)
   466→    if (dist < 58 && dist < bd) {
   467→      bd = dist
   468→      best = { label: `Поговорить: ${e.name}`, npc: e }
   469→    }
   470→  }
   471→  for (const it of INTERACTABLES) {
   472→    if (ctx.removedProps.has(it.id)) continue
   473→    if (!evalCond(ctx, it.cond ?? '')) continue
   474→    const dist = Math.hypot(it.pos.x - ctx.player.pos.x, it.pos.y - ctx.player.pos.y)
   475→    if (dist < it.r + 14 && dist < bd) {
   476→      bd = dist
   477→      best = { label: `${it.prompt}: ${it.name}`, it }
   478→    }
   479→  }
   480→  return best
   481→}

   483→export function doInteract(ctx: EngineCtx) {
   484→  if (ctx.isFrozen() || ctx.tacticalPaused) return
   485→  const n = nearestInteraction(ctx)
   486→  if (!n) return
   487→  if (n.npc) openDialogue(ctx, n.npc.npc!.dialogue)
   488→  else if (n.it) runInteractable(ctx, n.it)
   489→}

   493→export function advanceQuest(ctx: EngineCtx, id: string, stage: number) {
   494→  const q = ctx.quests[id]
   495→  if (!q || q.done) return
   496→  q.stage = stage
   497→  const def = QUEST_MAP[id]
   498→  if (stage >= def.stages.length) {
   499→    q.done = true
   500→    approvalOnQuestDone(ctx) // Фаза 4: завершение квеста — доверие компаньонов +6
   501→  }
   502→  if (id === 'main' && stage === 5 && !ctx.flags.gateOpen) {
   503→    ctx.flags.gateOpen = true
   504→    ctx.removedProps.add('barrier')
   505→    ctx.removedProps.add('barrier2')
   506→    ctx.removedProps.add('barrier3')
   507→    ctx.toast('Пелена перед Разломом опала!', 'quest')
   508→  }
   509→  // Фаза 6: главный квест Гранита — стадия 3 открывает выход Мастера Ямы
   510→  if (id === 'granite' && stage === 3 && !ctx.flags.pitMasterRevealed) {
   511→    ctx.flags.pitMasterRevealed = true
   512→    spawnPitMaster(ctx)
   513→    ctx.toast('Заслон Ямы гремит: Мастер ждёт тебя на арене.', 'warn')
   514→  }
   515→  ctx.toast(`${def.name}: ${stageText(ctx, id)}`, 'quest')
   516→  sfx('quest')
   517→  saveGame(ctx, false)
   518→  ctx.syncUi()
   519→}

   521→export function stageText(ctx: EngineCtx, id: string) {
   522→  const def = QUEST_MAP[id]
   523→  const q = ctx.quests[id]
   524→  if (q.done) return def.stages[def.stages.length - 1].text
   525→  const st = def.stages[Math.max(0, Math.min(def.stages.length - 1, q.stage - 1))]
   526→  return st ? st.text : ''
   527→}

   538→export function questMarker(ctx: EngineCtx): Vec | null {
   539→  const q = ctx.quests.main
   540→  if (!q.done) {
   541→    if (q.stage <= 1) return NPCS.find((n) => n.id === 'bjorn')!.pos
   542→    if (q.stage === 2) return { x: 900, y: 720 }
   543→    if (q.stage === 3) return { x: 2200, y: 660 }
   544→    if (q.stage === 4) {
   545→      if (!ctx.flags.alphaDead) return { x: 720, y: 450 }
   546→      if (!ctx.flags.captainDead) return { x: 2590, y: 1690 }
   547→      return { x: 1600, y: 2160 }
   548→    }
   549→    if (q.stage === 5) return { x: 2935, y: 470 }
   550→  }
   551→  const c = ctx.quests.contract
   552→  if (c.stage > 0 && !c.done) {
   553→    if (c.stage === 1) return { x: 720, y: 450 }
   554→    return NPCS.find((n) => n.id === 'grim')!.pos
   555→  }
   556→  // Контракт Вельда (Фаза 3)
   557→  const v = ctx.quests.veld
   558→  if (v.stage > 0 && !v.done) {
   559→    if (v.stage === 1) {
   560→      const clue = VELD_CLUES.find((cl) => !ctx.flags['clue_' + cl.id])
   561→      if (clue) return { ...clue.pos }
   562→      return { ...ASH_MOTHER.pos }
   563→    }
   564→    if (v.stage === 2) return { ...ASH_MOTHER.pos }
   565→    if (v.stage === 3) return NPCS.find((n) => n.id === 'alra')!.pos
   566→  }
   567→  // Гранитная Долина (Фаза 6)
   568→  const g = ctx.quests.granite
   569→  if (g.stage > 0 && !g.done) {
   570→    if (g.stage === 1) return NPCS.find((n) => n.id === 'kharan')!.pos
   571→    if (g.stage === 2) return NPCS.find((n) => n.id === 'kassia')!.pos
   572→    if (g.stage === 3) return { x: 4380, y: 2050 }
   573→    if (g.stage === 4) return NPCS.find((n) => n.id === 'kharan')!.pos
   574→  }
   575→  const go = ctx.quests.granite_ore
   576→  if (go.stage > 0 && !go.done) {
   577→    if (go.stage === 1) {
   578→      const vein = [
   579→        { id: 'ore1', pos: { x: 4020, y: 1050 } },
   580→        { id: 'ore2', pos: { x: 4560, y: 1020 } },
   581→        { id: 'ore3', pos: { x: 4790, y: 1450 } },
   582→        { id: 'ore4', pos: { x: 3950, y: 2350 } },
   583→        { id: 'ore5', pos: { x: 4560, y: 2680 } },
   584→      ].find((vv) => !ctx.flags['ore_' + vv.id])
   585→      if (vein) return { ...vein.pos }
   586→    }
   587→    return NPCS.find((n) => n.id === 'brock')!.pos
   588→  }
   589→  const gs = ctx.quests.granite_scorpions
   590→  if (gs.stage > 0 && !gs.done) {
   591→    if (gs.stage === 1) return { x: 4020, y: 2600 }
   592→    return NPCS.find((n) => n.id === 'finn')!.pos
   593→  }
   594→  // Стылое Пределье (Фаза 6.2)
   595→  const fw = ctx.quests.frost_words
   596→  if (fw.stage > 0 && !fw.done) {
   597→    if (fw.stage < 3) return { ...RUNE_CAVE.pos }
   598→    return { ...WORD_WALL_FORCE.pos }
   599→  }
   600→  const fd = ctx.quests.frost_dragon
   601→  if (fd.stage > 0 && !fd.done) {
   602→    if (fd.stage === 1) {
   603→      const clue = FROST_CLUES.find((cl) => !ctx.flags['clue_' + cl.id])
   604→      if (clue) return { ...clue.pos }
   605→      return { ...ROOST.pos }
   606→    }
   607→    if (fd.stage === 2 || fd.stage === 3) return { ...ROOST.pos }
   608→    return NPCS.find((n) => n.id === 'ingrid')!.pos
   609→  }
   610→  return null
   611→}

   613→export function updateArea(ctx: EngineCtx) {
   614→  // Фолбэк по зоне (Фаза 3/6/6.2): юг за Хребтом — Вельд, восток за Каменным Хребтом — Гранит,
   615→  // восток за Стылым Хребтом — Стылое Пределье
   616→  let name = ctx.zone === 'veld' ? 'Вельд' : ctx.zone === 'granite' ? 'Гранитная Долина' : ctx.zone === 'frost' ? 'Стылое Пределье' : 'Долина'
   617→  for (const a of AREAS) {
   618→    if (a.id === 'wilds') continue
   619→    const [x, y, w, h] = a.rect
   620→    if (ctx.player.pos.x > x && ctx.player.pos.x < x + w && ctx.player.pos.y > y && ctx.player.pos.y < y + h) {
   621→      name = a.name
   622→      break
   623→    }
   624→  }
   625→  if (name !== ctx.currentArea) {
   626→    ctx.currentArea = name
   627→    useGame.getState().patch({ areaToast: { id: Date.now(), name } })
   628→  }
   629→}
