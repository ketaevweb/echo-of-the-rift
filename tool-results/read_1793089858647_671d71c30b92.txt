   265→  ctx.fillRect(0, 0, edge, WORLD.h)

   270→  ctx.fillStyle = eg2
   271→  ctx.fillRect(0, 0, WORLD.w, edge)
   272→  ctx.save()
   273→  ctx.translate(0, WORLD.h)
   274→  ctx.scale(1, -1)
   275→  ctx.fillStyle = eg2
   276→  ctx.fillRect(0, 0, WORLD.w, edge)
   277→  ctx.restore()
   278→
   279→  return c
   280→}

   282→// ---------------- Пропсы ----------------
   283→
   284→export function drawProp(ctx: CanvasRenderingContext2D, p: PropDef, t: number) {
   285→  const { pos } = p
   286→  switch (p.kind) {
   287→    case 'tree': {
   288→      const s = 0.85 + ((p.seed ?? 0) % 30) / 100
   289→      ctx.fillStyle = 'rgba(0,0,0,0.25)'
   290→      ctx.beginPath()
   291→      ctx.ellipse(pos.x, pos.y + 4, 16 * s, 6 * s, 0, 0, Math.PI * 2)
   292→      ctx.fill()
   293→      ctx.fillStyle = '#4a3628'
   294→      ctx.fillRect(pos.x - 3, pos.y - 12, 6, 14)
   295→      const layers = [
   296→        { w: 30, h: 34, y: -22 },
   297→        { w: 24, h: 30, y: -44 },
   298→        { w: 17, h: 26, y: -64 },
   299→      ]
   300→      layers.forEach((L, i) => {
   301→        ctx.fillStyle = i === 2 ? '#31523a' : '#26402c'
   302→        ctx.beginPath()
   303→        ctx.moveTo(pos.x - L.w * s, pos.y + L.y * s)
   304→        ctx.lineTo(pos.x + L.w * s, pos.y + L.y * s)
   305→        ctx.lineTo(pos.x, pos.y + (L.y - L.h) * s)
   306→        ctx.closePath()
   307→        ctx.fill()
   308→      })
   309→      // снежная шапка
   310→      ctx.fillStyle = 'rgba(225,230,220,0.5)'
   311→      ctx.beginPath()
   312→      ctx.moveTo(pos.x - 8 * s, pos.y - 74 * s)
   313→      ctx.lineTo(pos.x + 8 * s, pos.y - 74 * s)
   314→      ctx.lineTo(pos.x, pos.y - 88 * s)
   315→      ctx.closePath()
   316→      ctx.fill()
   317→      break
   318→    }
   319→    case 'rock': {
   320→      const r = p.r ?? 10
   321→      ctx.fillStyle = 'rgba(0,0,0,0.25)'
   322→      ctx.beginPath()
   323→      ctx.ellipse(pos.x, pos.y + 3, r * 1.2, r * 0.5, 0, 0, Math.PI * 2)
   324→      ctx.fill()
   325→      ctx.fillStyle = '#6a6a62'
   326→      ctx.beginPath()
   327→      ctx.moveTo(pos.x - r, pos.y + 2)
   328→      ctx.lineTo(pos.x - r * 0.5, pos.y - r * 1.1)
   329→      ctx.lineTo(pos.x + r * 0.6, pos.y - r * 0.9)
   330→      ctx.lineTo(pos.x + r, pos.y + 2)
   331→      ctx.closePath()
   332→      ctx.fill()
   333→      ctx.fillStyle = 'rgba(230,230,225,0.35)'
   334→      ctx.beginPath()
   335→      ctx.moveTo(pos.x - r * 0.5, pos.y - r * 1.1)
   336→      ctx.lineTo(pos.x + r * 0.2, pos.y - r * 0.4)
   337→      ctx.lineTo(pos.x - r * 0.2, pos.y - r * 0.2)
   338→      ctx.closePath()
   339→      ctx.fill()
   340→      break
   341→    }
   342→    case 'house': {
   343→      const w = p.w ?? 120
   344→      const h = p.h ?? 70
   345→      ctx.fillStyle = 'rgba(0,0,0,0.3)'
   346→      ctx.beginPath()
   347→      ctx.ellipse(pos.x, pos.y + h * 0.45, w * 0.6, 14, 0, 0, Math.PI * 2)
   348→      ctx.fill()
   349→      ctx.fillStyle = '#5a4632'
   350→      ctx.fillRect(pos.x - w / 2, pos.y - h / 2, w, h)
   351→      ctx.fillStyle = '#4a3826'
   352→      ctx.fillRect(pos.x - w / 2, pos.y - h / 2, w, 6)
   353→      ctx.fillStyle = '#7a3b2e'
   354→      ctx.beginPath()
   355→      ctx.moveTo(pos.x - w / 2 - 10, pos.y - h / 2)
   356→      ctx.lineTo(pos.x + w / 2 + 10, pos.y - h / 2)
   357→      ctx.lineTo(pos.x + w * 0.2, pos.y - h / 2 - 34)
   358→      ctx.lineTo(pos.x - w * 0.2, pos.y - h / 2 - 34)
   359→      ctx.closePath()
   360→      ctx.fill()
   361→      ctx.strokeStyle = 'rgba(0,0,0,0.25)'
   362→      ctx.lineWidth = 2
   363→      ctx.stroke()
   364→      ctx.fillStyle = '#3a2c1c'
   365→      ctx.fillRect(pos.x - 10, pos.y + h / 2 - 26, 20, 26)
   366→      ctx.fillStyle = '#d9a441'
   367→      ctx.globalAlpha = 0.5 + 0.3 * Math.sin(t * 2 + pos.x)
   368→      ctx.fillRect(pos.x + w / 2 - 22, pos.y - 8, 10, 10)
   369→      ctx.globalAlpha = 1
   370→      break
   371→    }
   372→    case 'tent': {
   373→      const w = p.w ?? 60
   374→      ctx.fillStyle = 'rgba(0,0,0,0.25)'
   375→      ctx.beginPath()
   376→      ctx.ellipse(pos.x, pos.y + 12, w * 0.55, 9, 0, 0, Math.PI * 2)
   377→      ctx.fill()
   378→      ctx.fillStyle = '#6d5636'
   379→      ctx.beginPath()
   380→      ctx.moveTo(pos.x - w / 2, pos.y + 12)
   381→      ctx.lineTo(pos.x + w / 2, pos.y + 12)
   382→      ctx.lineTo(pos.x, pos.y - 34)
   383→      ctx.closePath()
   384→      ctx.fill()
   385→      ctx.fillStyle = '#59452a'
   386→      ctx.beginPath()
   387→      ctx.moveTo(pos.x, pos.y + 12)
   388→      ctx.lineTo(pos.x + w / 2, pos.y + 12)
   389→      ctx.lineTo(pos.x, pos.y - 34)
   390→      ctx.closePath()
   391→      ctx.fill()
   392→      break
   393→    }
   394→    case 'stone': {
   395→      const r = p.r ?? 12
   396→      ctx.fillStyle = 'rgba(0,0,0,0.3)'
   397→      ctx.beginPath()
   398→      ctx.ellipse(pos.x, pos.y + 3, r * 1.1, r * 0.45, 0, 0, Math.PI * 2)
   399→      ctx.fill()
   400→      ctx.fillStyle = '#70706a'
   401→      ctx.fillRect(pos.x - r * 0.7, pos.y - r * 2.6, r * 1.4, r * 2.6)
   402→      ctx.fillStyle = '#5a5a54'
   403→      ctx.fillRect(pos.x + r * 0.2, pos.y - r * 2.6, r * 0.5, r * 2.6)
   404→      ctx.fillStyle = `rgba(217,164,65,${0.45 + 0.25 * Math.sin(t * 2 + pos.x)})`
   405→      ctx.fillRect(pos.x - 2, pos.y - r * 2.1, 4, 10)
   406→      break
   407→    }

   859→      ctx.lineTo(pos.x + 11, pos.y - 6)

   870→    }
   871→    case 'bastion': {
   872→      // Фаза 6.3: ворота Бастиона — частокол с башнями и проёмом

   914→    case 'blight': {

   933→  }

   952→function drawWeaponSwing(ctx: CanvasRenderingContext2D, e: Entity, color: string) {
   953→  if (e.attackT <= 0) return
   954→  const p = 1 - e.attackT / 0.28
   955→  const a0 = e.facing - 1.1 + p * 2.2
   956→  ctx.save()
   957→  ctx.translate(e.pos.x, e.pos.y - 6)
   958→  ctx.rotate(a0)
   959→  ctx.strokeStyle = color
   960→  ctx.lineWidth = 3.5
   961→  ctx.lineCap = 'round'
   962→  ctx.beginPath()
   963→  ctx.moveTo(e.radius * 0.4, 0)
   964→  ctx.lineTo(e.radius * 1.5, 0)
   965→  ctx.stroke()
   966→  ctx.restore()
   967→  ctx.strokeStyle = `rgba(255,240,200,${0.35 * (1 - p)})`
   968→  ctx.lineWidth = 2
   969→  ctx.beginPath()
   970→  ctx.arc(e.pos.x, e.pos.y - 6, e.radius * 1.35, e.facing - 0.7, e.facing + 0.7)
   971→  ctx.stroke()
   972→}

   974→function drawHumanoid(
   975→  ctx: CanvasRenderingContext2D,
   976→  e: Entity,
   977→  t: number,
   978→  cloak: string,
   979→  weapon: 'sword' | 'staff' | 'shield' | 'bow' | 'none',
   980→) {
   981→  const bob = Math.sin(e.walkPhase * 10) * (Math.hypot(e.vel.x, e.vel.y) > 4 ? 1.6 : 0.4)
   982→  shadowBlob(ctx, e.pos.x, e.pos.y, e.radius)
   983→  const bx = e.pos.x
   984→  const by = e.pos.y - 8 + bob
   985→  // тело-плащ
   986→  ctx.fillStyle = e.hitT > 0 ? '#ffffff' : cloak
   987→  ctx.beginPath()
   988→  ctx.ellipse(bx, by, e.radius * 0.78, e.radius * 0.95, 0, 0, Math.PI * 2)
   989→  ctx.fill()
   990→  ctx.fillStyle = 'rgba(0,0,0,0.18)'
   991→  ctx.beginPath()
   992→  ctx.ellipse(bx - e.radius * 0.25, by + 3, e.radius * 0.4, e.radius * 0.7, 0, 0, Math.PI * 2)
   993→  ctx.fill()
   994→  // голова
   995→  ctx.fillStyle = '#d8b088'
   996→  ctx.beginPath()
   997→  ctx.arc(bx, by - e.radius * 0.95, e.radius * 0.5, 0, Math.PI * 2)
   998→  ctx.fill()
   999→  ctx.fillStyle = '#3a2c1c'
  1000→  ctx.beginPath()
  1001→  ctx.arc(bx, by - e.radius * 1.15, e.radius * 0.42, Math.PI, Math.PI * 2)
  1002→  ctx.fill()
  1003→  // оружие
  1004→  const fx = Math.cos(e.facing)
  1005→  const fy = Math.sin(e.facing)
  1006→  if (weapon === 'shield') {
  1007→    ctx.fillStyle = '#7a8a4a'
  1008→    ctx.beginPath()
  1009→    ctx.ellipse(bx + fx * e.radius * 0.9 - fy * e.radius * 0.5, by + fy * e.radius * 0.9 + fx * e.radius * 0.5, e.radius * 0.45, e.radius * 0.6, e.facing, 0, Math.PI * 2)
  1010→    ctx.fill()
  1011→    ctx.strokeStyle = '#4a522c'
  1012→    ctx.lineWidth = 2
  1013→    ctx.stroke()
  1014→    drawWeaponSwing(ctx, e, '#c9c4bc')
  1015→  } else if (weapon === 'staff') {
  1016→    ctx.strokeStyle = '#6a5638'
  1017→    ctx.lineWidth = 3
  1018→    ctx.beginPath()
  1019→    ctx.moveTo(bx - fy * e.radius * 0.7, by + fx * e.radius * 0.7)
  1020→    ctx.lineTo(bx - fy * e.radius * 0.7 + fx * 6, by + fx * e.radius * 0.7 + fy * 6 - e.radius * 1.6)
  1021→    ctx.stroke()
  1022→    const glow = 0.6 + 0.4 * Math.sin(t * 4 + e.eid)
  1023→    ctx.fillStyle = `rgba(159,184,164,${glow})`
  1024→    ctx.beginPath()
  1025→    ctx.arc(bx - fy * e.radius * 0.7 + fx * 6, by + fx * e.radius * 0.7 + fy * 6 - e.radius * 1.7, 3.5, 0, Math.PI * 2)
  1026→    ctx.fill()
  1027→  } else if (weapon === 'bow') {
  1028→    ctx.strokeStyle = '#8a7a54'
  1029→    ctx.lineWidth = 2.5
  1030→    ctx.beginPath()
  1031→    ctx.arc(bx + fx * e.radius, by + fy * e.radius, e.radius * 0.8, e.facing - 1.2, e.facing + 1.2)
  1032→    ctx.stroke()
  1033→  } else if (weapon === 'sword') {
  1034→    drawWeaponSwing(ctx, e, '#c9c4bc')
  1035→  }
  1036→  // щит Квена
  1037→  if (e.shield > 0) {
  1038→    ctx.strokeStyle = `rgba(217,164,65,${0.5 + 0.3 * Math.sin(t * 6)})`
  1039→    ctx.lineWidth = 2.5
  1040→    ctx.beginPath()
  1041→    ctx.arc(bx, by - 4, e.radius + 7, 0, Math.PI * 2)
  1042→    ctx.stroke()
  1043→  }
  1044→}

  1046→function drawWolf(ctx: CanvasRenderingContext2D, e: Entity, big: boolean) {
  1047→  const s = big ? 1.45 : 1
  1048→  const bob = Math.sin(e.walkPhase * 12) * (Math.hypot(e.vel.x, e.vel.y) > 4 ? 1.4 : 0.3)
  1049→  shadowBlob(ctx, e.pos.x, e.pos.y, e.radius * s)
  1050→  const bx = e.pos.x
  1051→  const by = e.pos.y - 7 * s + bob
  1052→  ctx.save()
  1053→  ctx.translate(bx, by)
  1054→  ctx.rotate(e.facing)
  1055→  ctx.fillStyle = e.hitT > 0 ? '#ffffff' : big ? '#7d7466' : '#565a5c'
  1056→  ctx.beginPath()
  1057→  ctx.ellipse(0, 0, e.radius * 1.25 * s, e.radius * 0.72 * s, 0, 0, Math.PI * 2)
  1058→  ctx.fill()
  1059→  // голова
  1060→  ctx.beginPath()
  1061→  ctx.ellipse(e.radius * 1.05 * s, -2 * s, e.radius * 0.55 * s, e.radius * 0.45 * s, 0, 0, Math.PI * 2)
  1062→  ctx.fill()
  1063→  // уши
  1064→  ctx.fillStyle = '#3d3a36'
  1065→  ctx.beginPath()
  1066→  ctx.moveTo(e.radius * 0.9 * s, -e.radius * 0.5 * s)
  1067→  ctx.lineTo(e.radius * 1.1 * s, -e.radius * 1.0 * s)
  1068→  ctx.lineTo(e.radius * 1.25 * s, -e.radius * 0.4 * s)
  1069→  ctx.closePath()
  1070→  ctx.fill()
  1071→  // хвост

  1073→  ctx.lineWidth = 4 * s
  1074→  ctx.beginPath()
  1075→  ctx.moveTo(-e.radius * 1.1 * s, 0)
  1076→  ctx.quadraticCurveTo(-e.radius * 1.7 * s, Math.sin(e.walkPhase * 9) * 4, -e.radius * 2 * s, -4 * s)

  1079→  ctx.fillStyle = big ? '#e0653a' : '#d9a441'
  1080→  ctx.beginPath()
  1081→  ctx.arc(e.radius * 1.35 * s, -4 * s, 1.8 * s, 0, Math.PI * 2)
  1082→  ctx.arc(e.radius * 1.35 * s, 1 * s, 1.8 * s, 0, Math.PI * 2)
  1083→  ctx.fill()
  1084→  ctx.restore()
  1085→}

  1116→function drawBoss(ctx: CanvasRenderingContext2D, e: Entity, t: number) {
  1117→  shadowBlob(ctx, e.pos.x, e.pos.y, e.radius * 1.2)
  1118→  const by = e.pos.y - 16 + Math.sin(t * 2) * 2
  1119→  const pulse = 0.5 + 0.5 * Math.sin(t * 5)
  1120→  // аура
  1121→  const g = ctx.createRadialGradient(e.pos.x, by, 6, e.pos.x, by, e.radius * 2.4)
  1122→  g.addColorStop(0, `rgba(160,107,217,${0.25 + pulse * 0.12})`)
  1123→  g.addColorStop(1, 'rgba(160,107,217,0)')
  1124→  ctx.fillStyle = g

  1145→  ctx.arc(e.pos.x, by, e.radius * 0.42 + pulse * 2, 0, Math.PI * 2)

  1147→  // глаза

  1153→}

  1155→export function drawEntity(
  1156→  ctx: CanvasRenderingContext2D,
  1157→  e: Entity,
  1158→  t: number,
  1159→  isTarget: boolean,
  1160→) {
  1161→  if (!e.alive) return
  1162→  if (isTarget) {
  1163→    ctx.strokeStyle = 'rgba(217,164,65,0.85)'
  1164→    ctx.lineWidth = 2
  1165→    ctx.setLineDash([6, 5])
  1166→    ctx.beginPath()
  1167→    ctx.ellipse(e.pos.x, e.pos.y + e.radius * 0.55, e.radius + 9, (e.radius + 9) * 0.45, 0, 0, Math.PI * 2)
  1168→    ctx.stroke()
  1169→    ctx.setLineDash([])
  1170→  }
  1171→  switch (e.enemyType) {
  1172→    case 'wolf':
  1173→      drawWolf(ctx, e, false)
  1174→      break
  1175→    case 'alpha':
  1176→      drawWolf(ctx, e, true)
  1177→      break
  1178→    case 'horror':
  1179→      drawHorror(ctx, e, t)
  1180→      break
  1181→    case 'draugr':
  1182→      // Пределье 2D: мёртвый воин в истлевшей броне
  1183→      drawHumanoid(ctx, e, t, '#6a7a6a', 'sword')
  1184→      break
  1185→    case 'ghoul':
  1186→      // Мара 2D: сгорбленный упырь в багровой корке
  1187→      drawHumanoid(ctx, e, t, '#8a5a4a', 'sword')
  1188→      break
  1189→    case 'abomination':
  1190→      // Мара 2D: сросшаяся громила
  1191→      drawHumanoid(ctx, e, t, '#6a4238', 'sword')
  1192→      break
  1193→    case 'blight_mother': {
  1194→      // Мора-Мать 2D: брющной мешок с щупальцами и ядром
  1195→      const bob = Math.sin(t * 2.1 + e.eid) * 2
  1196→      shadowBlob(ctx, e.pos.x, e.pos.y, e.radius * 1.2)
  1197→      ctx.fillStyle = e.hitT > 0 ? '#ffffff' : '#4a1610'
  1198→      ctx.beginPath()
  1199→      ctx.ellipse(e.pos.x, e.pos.y - e.radius * 0.7 - bob, e.radius * 1.25, e.radius * 0.95, 0, 0, Math.PI * 2)
  1200→      ctx.fill()
  1201→      // щупальца
  1202→      ctx.strokeStyle = e.hitT > 0 ? '#ffffff' : '#3a1210'
  1203→      ctx.lineWidth = 4
  1204→      for (let i = 0; i < 5; i++) {
  1205→        const a = t * 1.2 + (i / 5) * Math.PI * 2 + e.eid
  1206→        ctx.beginPath()
  1207→        ctx.moveTo(e.pos.x + Math.cos(a) * e.radius * 0.8, e.pos.y - 6 - bob)
  1208→        ctx.quadraticCurveTo(
  1209→          e.pos.x + Math.cos(a) * e.radius * 1.5,
  1210→          e.pos.y + 2,
  1211→          e.pos.x + Math.cos(a) * e.radius * 1.7,
  1212→          e.pos.y + 8,
  1213→        )
  1214→        ctx.stroke()
  1215→      }
  1216→      // ядро скверны
  1217→      const pulse = 0.5 + 0.5 * Math.sin(t * 4 + e.eid)
  1218→      ctx.fillStyle = `rgba(217,74,48,${0.6 + pulse * 0.35})`
  1219→      ctx.beginPath()
  1220→      ctx.arc(e.pos.x + e.radius * 0.4, e.pos.y - e.radius * 0.6 - bob, 5 + pulse * 1.5, 0, Math.PI * 2)
  1221→      ctx.fill()
  1222→      ctx.fillStyle = '#ffb04a'
  1223→      ctx.beginPath()
  1224→      ctx.arc(e.pos.x + e.radius * 1.05, e.pos.y - e.radius * 1.1 - bob, 2, 0, Math.PI * 2)
  1225→      ctx.arc(e.pos.x + e.radius * 1.05, e.pos.y - e.radius * 0.6 - bob, 2, 0, Math.PI * 2)
  1226→      ctx.fill()
  1227→      break
  1228→    }
  1229→    case 'frost_troll':
  1230→      // Стылый тролль: белошёрстная громила
  1231→      drawHumanoid(ctx, e, t, '#d8dcd4', 'sword')
  1232→      break
  1233→    case 'dragon': {
  1234→      // Иммолтакс 2D: крылатая тень в полёте/на земле
  1235→      const alt = e.cds['alt'] ?? 0
  1236→      shadowBlob(ctx, e.pos.x, e.pos.y, e.radius * (1 + alt * 0.35))
  1237→      ctx.save()
  1238→      ctx.translate(e.pos.x, e.pos.y - 10 - alt * 46)
  1239→      ctx.rotate(e.facing)
  1240→      const flap = Math.sin(t * (alt > 0.5 ? 7.5 : 1.4)) * (alt > 0.5 ? 0.5 : 0.08)
  1241→      // крылья
  1242→      ctx.fillStyle = e.hitT > 0 ? '#ffffff' : '#55322c'
  1243→      for (const sz of [-1, 1]) {
  1244→        ctx.save()
  1245→        ctx.rotate(sz * flap)
  1246→        ctx.beginPath()
  1247→        ctx.moveTo(-6, sz * 4)
  1248→        ctx.quadraticCurveTo(2, sz * 26, 16, sz * 20)
  1249→        ctx.quadraticCurveTo(6, sz * 12, 10, sz * 4)
  1250→        ctx.closePath()
  1251→        ctx.fill()
  1252→        ctx.restore()
  1253→      }
  1254→      // корпус
  1255→      ctx.fillStyle = e.hitT > 0 ? '#ffffff' : '#4a3230'
  1256→      ctx.beginPath()
  1257→      ctx.ellipse(0, 0, e.radius * 1.25, e.radius * 0.62, 0, 0, Math.PI * 2)
  1258→      ctx.fill()
  1259→      // голова/хвост
  1260→      ctx.beginPath()
  1261→      ctx.ellipse(e.radius * 1.3, -e.radius * 0.3, e.radius * 0.5, e.radius * 0.32, -0.2, 0, Math.PI * 2)
  1262→      ctx.fill()
  1263→      ctx.fillStyle = '#33221f'
  1264→      ctx.beginPath()
  1265→      ctx.ellipse(-e.radius * 1.5, e.radius * 0.1, e.radius * 0.9, e.radius * 0.24, 0.1, 0, Math.PI * 2)
  1266→      ctx.fill()
  1267→      // угли в пасти
  1268→      ctx.fillStyle = '#ffb04a'
  1269→      ctx.beginPath()
  1270→      ctx.arc(e.radius * 1.7, -e.radius * 0.3, 2, 0, Math.PI * 2)
  1271→      ctx.fill()
  1272→      ctx.restore()
  1273→      break
  1274→    }
  1275→    case 'bogling': {
  1276→      // Вельд 2D: присосистый ком тины
  1277→      shadowBlob(ctx, e.pos.x, e.pos.y, e.radius)
  1278→      const hop = Math.abs(Math.sin(e.walkPhase * 10)) * 3
  1279→      ctx.fillStyle = e.hitT > 0 ? '#ffffff' : '#4a5230'
  1280→      ctx.beginPath()
  1281→      ctx.ellipse(e.pos.x, e.pos.y - 5 - hop, e.radius, e.radius * 0.8, 0, 0, Math.PI * 2)
  1282→      ctx.fill()
  1283→      ctx.fillStyle = '#2c3220'
  1284→      ctx.beginPath()
  1285→      ctx.ellipse(e.pos.x + e.radius * 0.45, e.pos.y - 4 - hop, e.radius * 0.4, e.radius * 0.22, 0, 0, Math.PI * 2)
  1286→      ctx.fill()
  1287→      ctx.fillStyle = '#e8d568'
  1288→      ctx.beginPath()
  1289→      ctx.arc(e.pos.x + e.radius * 0.4, e.pos.y - 8 - hop, 1.8, 0, Math.PI * 2)
  1290→      ctx.arc(e.pos.x + e.radius * 0.4, e.pos.y - 4 - hop, 1.8, 0, Math.PI * 2)
  1291→      ctx.fill()
  1292→      break
  1293→    }
  1294→    case 'ash_shade':
  1295→    case 'ash_mother': {
  1296→      // Пепельные: парящий обугленный силуэт с углями
  1297→      const big = e.enemyType === 'ash_mother'
  1298→      const hover = Math.sin(t * 2.2 + e.eid) * 2
  1299→      shadowBlob(ctx, e.pos.x, e.pos.y, e.radius)
  1300→      ctx.fillStyle = e.hitT > 0 ? '#ffffff' : 'rgba(48,42,38,0.92)'
  1301→      ctx.beginPath()
  1302→      ctx.moveTo(e.pos.x - e.radius * (big ? 1.1 : 0.85), e.pos.y)
  1303→      ctx.quadraticCurveTo(e.pos.x - e.radius * (big ? 0.7 : 0.5), e.pos.y - e.radius * (big ? 3.4 : 2.4) - hover, e.pos.x, e.pos.y - e.radius * (big ? 3.8 : 2.7) - hover)
  1304→      ctx.quadraticCurveTo(e.pos.x + e.radius * (big ? 0.7 : 0.5), e.pos.y - e.radius * (big ? 3.4 : 2.4) - hover, e.pos.x + e.radius * (big ? 1.1 : 0.85), e.pos.y)
  1305→      ctx.closePath()
  1306→      ctx.fill()
  1307→      const pulse = 0.5 + 0.5 * Math.sin(t * 4 + e.eid)
  1308→      ctx.fillStyle = `rgba(255,122,48,${0.5 + pulse * 0.4})`
  1309→      ctx.beginPath()
  1310→      ctx.arc(e.pos.x, e.pos.y - e.radius * (big ? 1.6 : 1.2) - hover, big ? 5 : 2.6, 0, Math.PI * 2)
  1311→      ctx.fill()
  1312→      ctx.fillStyle = '#ffb04a'
  1313→      ctx.beginPath()
  1314→      ctx.arc(e.pos.x - 3, e.pos.y - e.radius * (big ? 2.9 : 2.2) - hover, 1.6, 0, Math.PI * 2)
  1315→      ctx.arc(e.pos.x + 3, e.pos.y - e.radius * (big ? 2.9 : 2.2) - hover, 1.6, 0, Math.PI * 2)
  1316→      ctx.fill()
  1317→      break
  1318→    }
  1319→    case 'boss':
  1320→      drawBoss(ctx, e, t)
  1321→      break
  1322→    case 'scorpion': {
  1323→      // Гранитный скорпион: панцирь, клешни, хвост-коса
  1324→      const bob = Math.sin(e.walkPhase * 11) * (Math.hypot(e.vel.x, e.vel.y) > 4 ? 1.2 : 0.3)
  1325→      shadowBlob(ctx, e.pos.x, e.pos.y, e.radius)
  1326→      ctx.save()
  1327→      ctx.translate(e.pos.x, e.pos.y - 6 + bob)
  1328→      ctx.rotate(e.facing)
  1329→      // хвост над спиной (дуга назад-вверх)
  1330→      ctx.strokeStyle = e.hitT > 0 ? '#ffffff' : '#6a6154'
  1331→      ctx.lineWidth = 4
  1332→      ctx.beginPath()
  1333→      ctx.moveTo(-e.radius * 0.8, 0)
  1334→      ctx.quadraticCurveTo(-e.radius * 1.9, -10 - bob * 2, -e.radius * 1.1, -18 - bob * 3)
  1335→      ctx.stroke()
  1336→      // жало
  1337→      ctx.fillStyle = '#d9a441'
  1338→      ctx.beginPath()
  1339→      ctx.arc(-e.radius * 1.1, -19 - bob * 3, 2.6, 0, Math.PI * 2)
  1340→      ctx.fill()
  1341→      // панцирь (сегменты)
  1342→      ctx.fillStyle = e.hitT > 0 ? '#ffffff' : '#7a7160'
  1343→      ctx.beginPath()
  1344→      ctx.ellipse(0, 0, e.radius * 1.1, e.radius * 0.72, 0, 0, Math.PI * 2)
  1345→      ctx.fill()
  1346→      ctx.fillStyle = '#5c544a'
  1347→      ctx.beginPath()
  1348→      ctx.ellipse(-e.radius * 0.55, 0, e.radius * 0.5, e.radius * 0.55, 0, 0, Math.PI * 2)
  1349→      ctx.fill()
  1350→      // клешни
  1351→      ctx.fillStyle = '#8a8070'
  1352→      ctx.beginPath()
  1353→      ctx.ellipse(e.radius * 1.15, -e.radius * 0.55, e.radius * 0.42, e.radius * 0.28, 0.5, 0, Math.PI * 2)
  1354→      ctx.fill()
  1355→      ctx.beginPath()
  1356→      ctx.ellipse(e.radius * 1.15, e.radius * 0.55, e.radius * 0.42, e.radius * 0.28, -0.5, 0, Math.PI * 2)
  1357→      ctx.fill()
  1358→      // глаза
  1359→      ctx.fillStyle = '#d9a441'
  1360→      ctx.beginPath()
  1361→      ctx.arc(e.radius * 0.55, -2.5, 1.6, 0, Math.PI * 2)
  1362→      ctx.arc(e.radius * 0.55, 2.5, 1.6, 0, Math.PI * 2)
  1363→      ctx.fill()
  1364→      ctx.restore()
  1365→      break
  1366→    }
  1367→    default: {
  1368→      const weapon =
  1369→        e.id === 'kaya'
  1370→          ? 'staff'
  1371→          : e.id === 'torvald'
  1372→            ? 'shield'
  1373→            : e.enemyType === 'archer'
  1374→              ? 'bow'
  1375→              : e.enemyType
  1376→                ? 'sword'
  1377→                : e.role === 'Архимаг Ковена' || e.role === 'Магистр-странница'
  1378→                  ? 'staff'
  1379→                  : 'sword'
  1380→      drawHumanoid(ctx, e, t, e.color, weapon)
  1381→    }
  1382→  }
  1383→  // оглушение
  1384→  if (e.stunT > 0) {
  1385→    for (let i = 0; i < 3; i++) {
  1386→      const a = t * 5 + (i / 3) * Math.PI * 2
  1387→      ctx.fillStyle = '#f5c96b'
  1388→      ctx.beginPath()
  1389→      ctx.arc(e.pos.x + Math.cos(a) * 10, e.pos.y - e.radius * 2.4 + Math.sin(a) * 4, 2.2, 0, Math.PI * 2)
  1390→      ctx.fill()
  1391→    }
  1392→  }
  1393→  // горение
  1394→  if (e.burnT > 0) {
  1395→    for (let i = 0; i < 2; i++) {
  1396→      const px = e.pos.x + Math.sin(t * 12 + i * 2 + e.eid) * 6
  1397→      ctx.fillStyle = `rgba(224,138,58,${0.5 + Math.random() * 0.3})`
  1398→      ctx.beginPath()
  1399→      ctx.arc(px, e.pos.y - 6 - Math.random() * 14, 2.4, 0, Math.PI * 2)
  1400→      ctx.fill()
  1401→    }
  1402→  }
  1403→  // замедление
  1404→  if (e.slowT > 0) {
  1405→    ctx.strokeStyle = 'rgba(200,230,240,0.5)'
  1406→    ctx.lineWidth = 1.5
  1407→    ctx.beginPath()
  1408→    ctx.arc(e.pos.x, e.pos.y, e.radius + 4, 0, Math.PI * 2)
  1409→    ctx.stroke()
  1410→  }
  1411→  // каст
  1412→  if (e.castT > 0) {
  1413→    ctx.strokeStyle = 'rgba(159,184,164,0.6)'
  1414→    ctx.lineWidth = 2
  1415→    ctx.beginPath()
  1416→    ctx.arc(e.pos.x, e.pos.y, e.radius + 10 - (0.4 - e.castT) * 20, 0, Math.PI * 2)
  1417→    ctx.stroke()
  1418→  }
  1419→  // полоса HP
  1420→  if (e.kind === 'enemy' && e.hp < e.maxHp) {
  1421→    hpBar(ctx, e.pos.x, e.pos.y - e.radius * 2.2 - 8, Math.max(26, e.radius * 2), e.hp / e.maxHp, e.boss ? '#a06bd9' : '#c94f3d')
  1422→  }
  1423→  if (e.kind === 'companion' && e.hp < e.maxHp) {
  1424→    hpBar(ctx, e.pos.x, e.pos.y - e.radius * 2.2 - 8, 26, e.hp / e.maxHp, '#8fbf7f')
  1425→  }
  1426→  // имя
  1427→  if (e.kind === 'npc' || e.kind === 'companion') {
  1428→    ctx.font = '11px ui-sans-serif, system-ui'
  1429→    ctx.textAlign = 'center'
  1430→    ctx.fillStyle = 'rgba(0,0,0,0.55)'
  1431→    const tw = ctx.measureText(e.name).width
  1432→    ctx.fillRect(e.pos.x - tw / 2 - 4, e.pos.y - e.radius * 2.2 - 24, tw + 8, 14)
  1433→    ctx.fillStyle = e.kind === 'companion' ? '#c9e0b8' : '#e8d5a8'
  1434→    ctx.fillText(e.name, e.pos.x, e.pos.y - e.radius * 2.2 - 13)
  1435→  }
  1436→}

  1440→export function drawParticles(ctx: CanvasRenderingContext2D, ps: Particle[]) {
  1441→  for (const p of ps) {
  1442→    const a = p.life / p.maxLife
  1443→    ctx.globalAlpha = Math.max(0, a)
  1444→    if (p.kind === 'ring' || p.kind === 'slam') {
  1445→      ctx.strokeStyle = p.color
  1446→      ctx.lineWidth = p.kind === 'slam' ? 4 : 2.5
  1447→      ctx.beginPath()
  1448→      ctx.arc(p.x, p.y, p.size * (1 - a) * 2.2 + 6, 0, Math.PI * 2)
  1449→      ctx.stroke()
  1450→    } else {
  1451→      ctx.fillStyle = p.color
  1452→      ctx.beginPath()
  1453→      ctx.arc(p.x, p.y, p.size * (0.4 + a * 0.6), 0, Math.PI * 2)
  1454→      ctx.fill()
  1455→    }
  1456→  }
  1457→  ctx.globalAlpha = 1
  1458→}

  1470→    ctx.fillText(f.text, f.x, f.y)
  1471→  }

  1477→export function drawMinimap(
  1478→  mini: HTMLCanvasElement,
  1479→  baked: HTMLCanvasElement,
  1480→  entities: Entity[],
  1481→  playerPos: Vec,
  1482→  questMarker: Vec | null,
  1483→  size: number,
  1484→  t: number,
  1485→) {
  1486→  const ctx = mini.getContext('2d')
  1487→  if (!ctx) return
  1488→  const scale = size / WORLD.w
  1489→  ctx.clearRect(0, 0, size, size)
  1490→  ctx.drawImage(baked, 0, 0, WORLD.w, WORLD.h, 0, 0, size, size * (WORLD.h / WORLD.w))
  1491→  const mh = size * (WORLD.h / WORLD.w)
  1492→  const dot = (x: number, y: number, color: string, r: number) => {
  1493→    ctx.fillStyle = color
  1494→    ctx.beginPath()
  1495→    ctx.arc(x * scale, y * (mh / WORLD.h), r, 0, Math.PI * 2)
  1496→    ctx.fill()
  1497→  }
  1498→  for (const e of entities) {
  1499→    if (!e.alive) continue
  1500→    if (e.kind === 'npc') dot(e.pos.x, e.pos.y, '#e8d5a8', 2)
  1501→    else if (e.kind === 'companion') dot(e.pos.x, e.pos.y, '#c9e0b8', 2)
  1502→    else if (e.kind === 'enemy' && e.aiState !== 'return') dot(e.pos.x, e.pos.y, '#c94f3d', 2)
  1503→  }
  1504→  if (questMarker) {
  1505→    const pulse = 2 + Math.sin(t * 4) * 1
  1506→    ctx.strokeStyle = '#f5c96b'
  1507→    ctx.lineWidth = 1.5
  1508→    ctx.beginPath()
  1509→    ctx.arc(questMarker.x * scale, questMarker.y * (mh / WORLD.h), 4 + pulse, 0, Math.PI * 2)
  1510→    ctx.stroke()
  1511→  }
  1512→  // игрок
  1513→  ctx.fillStyle = '#f5c96b'
  1514→  ctx.beginPath()
  1515→  ctx.arc(playerPos.x * scale, playerPos.y * (mh / WORLD.h), 3, 0, Math.PI * 2)
  1516→  ctx.fill()
  1517→  ctx.strokeStyle = 'rgba(255,255,255,0.7)'
  1518→  ctx.lineWidth = 1
  1519→  ctx.stroke()
  1520→}
