     8→import * as THREE from 'three'
     9→import { INTERACTABLES, PROPS } from '../data/world'
    10→import { PropDef } from '../types'
    11→import { seedRot } from '../core/util'
    12→import { heightAt } from './heightmap'
    13→
    14→const V = (x: number, y: number, z: number) => new THREE.Vector3(x, y, z)
    15→
    16→function groundMat(color: number, rough = 0.95): THREE.MeshStandardMaterial {
    17→  return new THREE.MeshStandardMaterial({ color, roughness: rough, metalness: 0 })
    18→}
    19→
    20→function place(group: THREE.Group, obj: THREE.Object3D, x: number, y: number, z = 0, rotY = 0) {
    21→  obj.position.set(x, heightAt(x, y) + z, y)
    22→  obj.rotation.y = rotY
    23→  group.add(obj)
    24→}
    25→
    26→export class Props3d {
    27→  group = new THREE.Group()
    28→  /** Меши, скрываемые при removedProps (барьеры, травы). */
    29→  private removable = new Map<string, THREE.Object3D[]>()
    30→  private fires: Array<{ light: THREE.PointLight; flame: THREE.Mesh; base: number }> = []
    31→  /** Вращающиеся магические элементы (руна алтаря, ядра порталов). */
    32→  private spinners: THREE.Object3D[] = []
    33→  private herbPulse: THREE.Object3D[] = []
    34→  /** Фаза 6.2: кольца рунных камней (обновляются цветом из состояния движка). */
    35→  private runeRings = new Map<string, THREE.Mesh>()
    36→  private disposables: Array<{ dispose(): void }> = []
    37→
    38→  constructor() {
    39→    this.buildInstanced()
    40→    for (const p of PROPS) this.buildProp(p)
    41→    this.buildHerbs()
    42→  }
    43→
    44→  // ---------------- Инстансинг: деревья/камни/кусты ----------------
    45→
    46→  private buildInstanced() {
    47→    const trees = PROPS.filter((p) => p.kind === 'tree')
    48→    const rocks = PROPS.filter((p) => p.kind === 'rock')
    49→    const shrubs = PROPS.filter((p) => p.kind === 'shrub' || p.kind === 'flower')
    50→    const deadTrees = PROPS.filter((p) => p.kind === 'deadtree')
    51→    const reeds = PROPS.filter((p) => p.kind === 'reed')
    52→    const snowTrees = PROPS.filter((p) => p.kind === 'snowtree')
    53→
    54→    if (trees.length) {
    55→      // Ствол
    56→      const trunkGeo = new THREE.CylinderGeometry(1.6, 2.6, 14, 6)
    57→      trunkGeo.translate(0, 7, 0)
    58→      const trunkMat = groundMat(0x4a3626)
    59→      const trunks = new THREE.InstancedMesh(trunkGeo, trunkMat, trees.length)
    60→      // Крона — два конуса (матрицы инстансов — МИРОВЫЕ: база дерева + локальный сдвиг * масштаб)
    61→      const coneGeo = new THREE.ConeGeometry(8, 18, 7)
    62→      const leafMat = groundMat(0x2e4422)
    63→      const leaves = new THREE.InstancedMesh(coneGeo, leafMat, trees.length * 2)
    64→      const m = new THREE.Matrix4()
    65→      const q = new THREE.Quaternion()
    66→      const sc = new THREE.Vector3()
    67→      trees.forEach((p, i) => {
    68→        const h = heightAt(p.pos.x, p.pos.y)
    69→        const rot = seedRot(p.seed ?? i)
    70→        const s = 0.85 + ((p.seed ?? 7) % 100) / 220
    71→        q.setFromAxisAngle(V(0, 1, 0), rot)
    72→        sc.set(s, s * (0.9 + ((p.seed ?? 3) % 60) / 200), s)
    73→        m.compose(V(p.pos.x, h, p.pos.y), q, sc)
    74→        trunks.setMatrixAt(i, m)
    75→        m.compose(V(p.pos.x, h + 12 * s, p.pos.y), q, sc)
    76→        leaves.setMatrixAt(i * 2, m)
    77→        m.compose(V(p.pos.x, h + 22 * s, p.pos.y), q, sc)
    78→        leaves.setMatrixAt(i * 2 + 1, m)
    79→      })
    80→      trunks.castShadow = true
    81→      leaves.castShadow = true
    82→      this.group.add(trunks, leaves)
    83→      this.disposables.push(trunkGeo, trunkMat, coneGeo, leafMat)
    84→    }
    85→
    86→    if (rocks.length) {
    87→      const rockGeo = new THREE.DodecahedronGeometry(1, 0)
    88→      const rockMat = groundMat(0x6a6a64)
    89→      const meshes = new THREE.InstancedMesh(rockGeo, rockMat, rocks.length)
    90→      const m = new THREE.Matrix4()
    91→      const q = new THREE.Quaternion()
    92→      const e = new THREE.Euler()
    93→      const sc = new THREE.Vector3()
    94→      rocks.forEach((p, i) => {
    95→        const h = heightAt(p.pos.x, p.pos.y)
    96→        const r = p.r ?? 10
    97→        e.set(seedRot(p.seed ?? i) * 0.4, seedRot((p.seed ?? 0) + 91), 0)
    98→        q.setFromEuler(e)
    99→        sc.set(r * 1.15, r * 0.8, r * 1.05)
   100→        m.compose(V(p.pos.x, h + r * 0.25, p.pos.y), q, sc)
   101→        meshes.setMatrixAt(i, m)
   102→      })
   103→      meshes.castShadow = true
   104→      meshes.receiveShadow = true
   105→      this.group.add(meshes)
   106→      this.disposables.push(rockGeo, rockMat)
   107→    }
   108→
   109→    if (shrubs.length) {
   110→      const bushGeo = new THREE.IcosahedronGeometry(1, 0)
   111→      const bushMat = groundMat(0x3c5226)
   112→      const flowerMat = groundMat(0xa8b86a)
   113→      const bushes = new THREE.InstancedMesh(bushGeo, bushMat, shrubs.length)
   114→      const m = new THREE.Matrix4()
   115→      const q = new THREE.Quaternion()
   116→      const sc = new THREE.Vector3()
   117→      shrubs.forEach((p, i) => {
   118→        const h = heightAt(p.pos.x, p.pos.y)
   119→        q.setFromAxisAngle(V(0, 1, 0), seedRot(p.seed ?? i))
   120→        const s = p.kind === 'flower' ? 1.5 : 2.4
   121→        sc.set(s, s * 0.75, s)
   122→        m.compose(V(p.pos.x, h + s * 0.5, p.pos.y), q, sc)
   123→        bushes.setMatrixAt(i, m)
   124→      })
   125→      this.group.add(bushes)
   126→      this.disposables.push(bushGeo, bushMat, flowerMat)
   127→    }
   128→
   129→    // Мёртвые деревья Вельда: голый ствол + две ветки (Фаза 3)
   130→    if (deadTrees.length) {
   131→      const trunkGeo = new THREE.CylinderGeometry(0.9, 2.2, 15, 6)
   132→      trunkGeo.translate(0, 7.5, 0)
   133→      const branchGeo = new THREE.CylinderGeometry(0.45, 0.8, 8, 5)
   134→      branchGeo.translate(0, 4, 0)
   135→      const barkMat = groundMat(0x3a322a, 0.98)
   136→      const trunks = new THREE.InstancedMesh(trunkGeo, barkMat, deadTrees.length)
   137→      const branches = new THREE.InstancedMesh(branchGeo, barkMat, deadTrees.length * 2)
   138→      const m = new THREE.Matrix4()
   139→      const q = new THREE.Quaternion()
   140→      const q2 = new THREE.Quaternion()
   141→      const e2 = new THREE.Euler()
   142→      const sc = new THREE.Vector3()
   143→      deadTrees.forEach((p, i) => {
   144→        const h = heightAt(p.pos.x, p.pos.y)
   145→        const rot = seedRot(p.seed ?? i)
   146→        const s = 0.8 + ((p.seed ?? 7) % 100) / 240
   147→        q.setFromAxisAngle(V(0, 1, 0), rot)
   148→        sc.set(s, s, s)
   149→        m.compose(V(p.pos.x, h, p.pos.y), q, sc)
   150→        trunks.setMatrixAt(i, m)
   151→        // Ветки: наклонены вбок от ствола
   152→        e2.set(((p.seed ?? 1) % 7) * 0.05 + 0.55, rot + 1.9, 0)
   153→        q2.setFromEuler(e2)
   154→        m.compose(V(p.pos.x, h + 9.5 * s, p.pos.y), q2, sc)
   155→        branches.setMatrixAt(i * 2, m)
   156→        e2.set(0.7, rot + 4.6, 0)
   157→        q2.setFromEuler(e2)
   158→        m.compose(V(p.pos.x, h + 7.5 * s, p.pos.y), q2, sc)
   159→        branches.setMatrixAt(i * 2 + 1, m)
   160→      })
   161→      trunks.castShadow = true
   162→      branches.castShadow = true
   163→      this.group.add(trunks, branches)
   164→      this.disposables.push(trunkGeo, branchGeo, barkMat)
   165→    }
   166→
   167→    // Камыш у омутов (Фаза 3)
   168→    if (reeds.length) {
   169→      const reedGeo = new THREE.ConeGeometry(0.9, 6.5, 4)
   170→      reedGeo.translate(0, 3.2, 0)
   171→      const reedMat = groundMat(0x4a5530, 0.98)
   172→      const mesh = new THREE.InstancedMesh(reedGeo, reedMat, reeds.length * 3)
   173→      const m = new THREE.Matrix4()
   174→      const q = new THREE.Quaternion()
   175→      const sc = new THREE.Vector3()
   176→      reeds.forEach((p, i) => {
   177→        const h = heightAt(p.pos.x, p.pos.y)
   178→        for (let k = 0; k < 3; k++) {
   179→          const ox = ((p.seed ?? i) % 13) - 6 + k * 3.1
   180→          const oz = (((p.seed ?? i) * 7) % 11) - 5
   181→          q.setFromAxisAngle(V(0, 1, 0), seedRot((p.seed ?? 0) + k * 31))
   182→          const s = 0.7 + ((p.seed ?? 3) % 50) / 90
   183→          sc.set(s, s * (0.85 + k * 0.12), s)
   184→          m.compose(V(p.pos.x + ox, h, p.pos.y + oz), q, sc)
   185→          mesh.setMatrixAt(i * 3 + k, m)
   186→        }
   187→      })
   188→      this.group.add(mesh)
   189→      this.disposables.push(reedGeo, reedMat)
   190→    }
   191→
   192→    // Снежные ели Пределья (Фаза 6.2): тёмная хвоя под шапкой снега
   193→    if (snowTrees.length) {
   194→      const trunkGeo = new THREE.CylinderGeometry(1.3, 2.2, 12, 6)
   195→      trunkGeo.translate(0, 6, 0)
   196→      const barkMat = groundMat(0x3c3226)
   197→      const trunks = new THREE.InstancedMesh(trunkGeo, barkMat, snowTrees.length)
   198→      const coneGeo = new THREE.ConeGeometry(6.5, 15, 7)
   199→      const needleMat = groundMat(0x2a3a28)
   200→      const snowMatI = groundMat(0xd8dee0, 0.92)
   201→      const cones = new THREE.InstancedMesh(coneGeo, needleMat, snowTrees.length * 2)
   202→      const caps = new THREE.InstancedMesh(new THREE.ConeGeometry(4.4, 5.5, 7), snowMatI, snowTrees.length)
   203→      const m = new THREE.Matrix4()
   204→      const q = new THREE.Quaternion()
   205→      const sc = new THREE.Vector3()
   206→      snowTrees.forEach((p, i) => {
   207→        const h = heightAt(p.pos.x, p.pos.y)
   208→        const rot = seedRot(p.seed ?? i)
   209→        const s = 0.85 + ((p.seed ?? 7) % 100) / 200
   210→        q.setFromAxisAngle(V(0, 1, 0), rot)
   211→        sc.set(s, s, s)
   212→        m.compose(V(p.pos.x, h, p.pos.y), q, sc)
   213→        trunks.setMatrixAt(i, m)
   214→        m.compose(V(p.pos.x, h + 11 * s, p.pos.y), q, sc)
   215→        cones.setMatrixAt(i * 2, m)
   216→        m.compose(V(p.pos.x, h + 19 * s, p.pos.y), q, sc)
   217→        cones.setMatrixAt(i * 2 + 1, m)
   218→        m.compose(V(p.pos.x, h + 26.5 * s, p.pos.y), q, sc)
   219→        caps.setMatrixAt(i, m)
   220→      })
   221→      trunks.castShadow = true
   222→      cones.castShadow = true
   223→      this.group.add(trunks, cones, caps)
   224→      this.disposables.push(trunkGeo, barkMat, coneGeo, needleMat, snowMatI)
   225→    }
   226→  }

   228→  // ---------------- Одиночные пропсы ----------------
   229→
   230→  private buildProp(p: PropDef) {
   231→    const { x, y } = p.pos
   232→    const h = () => heightAt(x, y)
   233→    switch (p.kind) {
   234→      case 'tree':
   235→      case 'rock':
   236→      case 'shrub':
   237→      case 'flower':
   238→        return // уже в инстансах
   239→
   240→      case 'house': {
   241→        const g = new THREE.Group()
   242→        const w = p.w ?? 120
   243→        const d = p.h ?? 70
   244→        const bodyH = 26 + w * 0.06
   245→        const body = new THREE.Mesh(new THREE.BoxGeometry(w, bodyH, d), groundMat(0x6b543c))
   246→        body.position.y = bodyH / 2
   247→        body.castShadow = true
   248→        body.receiveShadow = true
   249→        const roof = new THREE.Mesh(new THREE.ConeGeometry(Math.max(w, d) * 0.62, 20 + w * 0.08, 4), groundMat(0x3d2f22))
   250→        roof.rotation.y = Math.PI / 4
   251→        roof.position.y = bodyH + (10 + w * 0.04)
   252→        roof.castShadow = true
   253→        g.add(body, roof)
   254→        place(this.group, g, x, y, 0, seedRot(p.seed ?? 1) * 0.2)
   255→        break
   256→      }
   257→
   258→      case 'tent': {
   259→        const g = new THREE.Group()
   260→        const w = p.w ?? 60
   261→        const tent = new THREE.Mesh(new THREE.ConeGeometry(w * 0.42, w * 0.62, 5), groundMat(0x7a6248))
   262→        tent.position.y = w * 0.31
   263→        tent.castShadow = true
   264→        g.add(tent)
   265→        place(this.group, g, x, y, 0, seedRot(p.seed ?? 1))
   266→        break
   267→      }
   268→
   269→      case 'stall': {
   270→        const g = new THREE.Group()
   271→        const table = new THREE.Mesh(new THREE.BoxGeometry(70, 8, 42), groundMat(0x6b543c))
   272→        table.position.y = 16
   273→        const roof = new THREE.Mesh(new THREE.BoxGeometry(80, 4, 50), groundMat(0x8a3d2e))
   274→        roof.position.y = 40
   275→        roof.rotation.z = 0.08
   276→        for (const [lx, lz] of [[-30, -16], [30, -16], [-30, 16], [30, 16]]) {
   277→          const leg = new THREE.Mesh(new THREE.CylinderGeometry(1.6, 1.6, 32, 5), groundMat(0x4a3626))
   278→          leg.position.set(lx, 16, lz)
   279→          g.add(leg)
   280→        }
   281→        table.castShadow = true
   282→        g.add(table, roof)
   283→        place(this.group, g, x, y)
   284→        break
   285→      }
   286→
   287→      case 'dummy': {
   288→        const g = new THREE.Group()
   289→        const post = new THREE.Mesh(new THREE.CylinderGeometry(3, 4, 34, 6), groundMat(0x5c4630))
   290→        post.position.y = 17
   291→        const cross = new THREE.Mesh(new THREE.BoxGeometry(26, 4, 4), groundMat(0x5c4630))
   292→        cross.position.y = 26
   293→        const head = new THREE.Mesh(new THREE.SphereGeometry(4.5, 8, 6), groundMat(0x8a7148))
   294→        head.position.y = 37
   295→        post.castShadow = true
   296→        g.add(post, cross, head)
   297→        place(this.group, g, x, y)
   298→        break
   299→      }
   300→
   301→      case 'fire': {
   302→        const g = new THREE.Group()
   303→        for (let i = 0; i < 5; i++) {
   304→          const log = new THREE.Mesh(new THREE.CylinderGeometry(1.4, 1.8, 12, 5), groundMat(0x3d2a1a))
   305→          log.rotation.set(Math.PI / 2.4, (i / 5) * Math.PI * 2, 0)
   306→          log.position.y = 2.4
   307→          g.add(log)
   308→        }
   309→        const flame = new THREE.Mesh(
   310→          new THREE.ConeGeometry(5, 12, 6),
   311→          new THREE.MeshBasicMaterial({ color: 0xe8913a, transparent: true, opacity: 0.92 }),
   312→        )
   313→        flame.position.y = 9
   314→        const light = new THREE.PointLight(0xff9540, 320, 190, 2)
   315→        light.position.y = 12
   316→        g.add(flame, light)
   317→        place(this.group, g, x, y)
   318→        this.fires.push({ light, flame, base: 1 })
   319→        break
   320→      }
   321→
   322→      case 'fence': {
   323→        const g = new THREE.Group()
   324→        const w = p.w ?? 90
   325→        const rail = new THREE.Mesh(new THREE.BoxGeometry(w, 3, 2.4), groundMat(0x54422e))
   326→        rail.position.y = 12
   327→        const rail2 = rail.clone()
   328→        rail2.position.y = 6
   329→        g.add(rail, rail2)
   330→        const n = Math.max(2, Math.round(w / 30))
   331→        for (let i = 0; i <= n; i++) {
   332→          const post = new THREE.Mesh(new THREE.CylinderGeometry(1.6, 1.6, 14, 5), groundMat(0x4a3626))
   333→          post.position.set(-w / 2 + (w / n) * i, 7, 0)
   334→          g.add(post)
   335→        }
   336→        place(this.group, g, x, y, 0, seedRot(p.seed ?? 1) * 0.1)
   337→        break
   338→      }
   339→
   340→      case 'stone': {
   341→        const g = new THREE.Group()
   342→        const r = p.r ?? 12
   343→        const obelisk = new THREE.Mesh(new THREE.BoxGeometry(r * 0.9, r * 3.4, r * 0.7), groundMat(0x77776f))
   344→        obelisk.position.y = r * 1.6
   345→        obelisk.rotation.z = (seedRot(p.seed ?? 1) - Math.PI) * 0.06
   346→        obelisk.castShadow = true
   347→        const cap = new THREE.Mesh(new THREE.ConeGeometry(r * 0.55, r * 0.9, 4), groundMat(0x6a6a62))
   348→        cap.position.y = r * 3.4
   349→        cap.rotation.y = Math.PI / 4
   350→        g.add(obelisk, cap)
   351→        place(this.group, g, x, y)
   352→        break
   353→      }
   354→
   355→      case 'altar': {
   356→        const g = new THREE.Group()
   357→        const slab = new THREE.Mesh(new THREE.CylinderGeometry(16, 20, 8, 8), groundMat(0x8a8a80))
   358→        slab.position.y = 4
   359→        slab.castShadow = true
   360→        const glow = new THREE.Mesh(
   361→          new THREE.OctahedronGeometry(5, 0),
   362→          new THREE.MeshBasicMaterial({ color: 0xd9a441, transparent: true, opacity: 0.85 }),
   363→        )
   364→        glow.position.y = 22
   365→        const ring = new THREE.Mesh(
   366→          new THREE.RingGeometry(20, 23, 32),
   367→          new THREE.MeshBasicMaterial({ color: 0xd9a441, transparent: true, opacity: 0.3, side: THREE.DoubleSide }),
   368→        )
   369→        ring.rotation.x = -Math.PI / 2
   370→        ring.position.y = 1
   371→        g.add(slab, glow, ring)
   372→        place(this.group, g, x, y)
   373→        this.spinners.push(glow) // вращается вместе с руной алтаря
   374→        this.herbPulse.push(ring)
   375→        break
   376→      }
   377→
   378→      case 'carcass': {
   379→        const g = new THREE.Group()
   380→        const body = new THREE.Mesh(new THREE.CapsuleGeometry(p.r ?? 14, 10, 3, 8), groundMat(0x5e4438))
   381→        body.rotation.z = Math.PI / 2
   382→        body.position.y = 5
   383→        g.add(body)
   384→        place(this.group, g, x, y)
   385→        break
   386→      }
   387→
   388→      case 'rift': {
   389→        const g = new THREE.Group()
   390→        const portal = new THREE.Mesh(
   391→          new THREE.TorusGeometry(52, 7, 10, 40),
   392→          new THREE.MeshStandardMaterial({ color: 0x3a2a4a, emissive: 0x9a5fd0, emissiveIntensity: 1.4, roughness: 0.5 }),
   393→        )
   394→        portal.position.y = 62
   395→        const disc = new THREE.Mesh(
   396→          new THREE.CircleGeometry(50, 32),
   397→          new THREE.MeshBasicMaterial({ color: 0x1a0f22, transparent: true, opacity: 0.88, side: THREE.DoubleSide }),
   398→        )
   399→        disc.position.y = 62
   400→        const core = new THREE.Mesh(
   401→          new THREE.TorusGeometry(26, 4, 8, 28),
   402→          new THREE.MeshBasicMaterial({ color: 0xc09ae8, transparent: true, opacity: 0.7 }),
   403→        )
   404→        core.position.y = 62
   405→        const light = new THREE.PointLight(0xa86fe0, 900, 520, 2)
   406→        light.position.y = 70
   407→        g.add(portal, disc, core, light)
   408→        place(this.group, g, x, y)
   409→        this.spinners.push(core)
   410→        break
   411→      }
   412→
   413→      case 'barrier': {
   414→        const g = new THREE.Group()
   415→        const shroud = new THREE.Mesh(
   416→          new THREE.CylinderGeometry(20, 26, 46, 8, 1, true),
   417→          new THREE.MeshStandardMaterial({
   418→            color: 0x6a4a8a,
   419→            emissive: 0x8a5fc0,
   420→            emissiveIntensity: 0.7,
   421→            transparent: true,
   422→            opacity: 0.45,
   423→            side: THREE.DoubleSide,
   424→          }),
   425→        )
   426→        shroud.position.y = 23
   427→        g.add(shroud)
   428→        place(this.group, g, x, y)
   429→        if (p.id) this.removable.set(p.id, [g])
   430→        break
   431→      }
   432→
   433→      // --- Фаза 3: доска объявлений, врата Тумана, пень ---
   434→
   435→      case 'board': {
   436→        const g = new THREE.Group()
   437→        for (const px of [-16, 16]) {
   438→          const post = new THREE.Mesh(new THREE.CylinderGeometry(1.8, 2.2, 26, 6), groundMat(0x4a3626))

   872→  update(time: number, removed: Set<string>, darkness = 0, runes?: { rs1: number; rs2: number; rs3: number }) {
   873→    // Огни костров мерцают; ночью — ярче (Фаза 3)
   874→    const nightBoost = 0.8 + darkness * 1.5
   875→    for (const f of this.fires) {
   876→      const flick = 0.82 + Math.sin(time * 11) * 0.1 + Math.sin(time * 23.7) * 0.08
   877→      f.light.intensity = 320 * flick * nightBoost
   878→      const s = 0.9 + flick * 0.25
   879→      f.flame.scale.set(s, 1 + flick * 0.4, s)
   880→    }
   881→    // Пульс колец трав/алтаря
   882→    const pulse = 0.5 + 0.5 * Math.sin(time * 2.5)
   883→    for (const r of this.herbPulse) {
   884→      const m = (r as THREE.Mesh).material as THREE.MeshBasicMaterial
   885→      m.opacity = 0.22 + pulse * 0.3
   886→    }
   887→    // Вращение магических элементов
   888→    for (const s of this.spinners) {
   889→      s.rotation.y = time * 0.9
   890→      s.rotation.z = Math.sin(time * 0.6) * 0.15
   891→    }
   892→    // Фаза 6.2: кольца рунных камней перекрашиваются по состоянию головоломки
   893→    if (runes) {
   894→      const colors = [0x9fb8a4, 0xd9a441, 0xd8dde2]
   895→      for (const [id, mesh] of this.runeRings) {
   896→        const idx = runes[id as 'rs1'] ?? 0
   897→        const mat = mesh.material as THREE.MeshStandardMaterial
   898→        mat.color.setHex(colors[idx])
   899→        mat.emissive.setHex(colors[idx])
   900→      }
   901→    }
   902→    // Скрытые пропсы (барьеры по квесту, собранные травы)
   903→    for (const [id, objs] of this.removable) {
   904→      const hidden = removed.has(id)
   905→      for (const o of objs) o.visible = !hidden
   906→    }
   907→  }
   908→
   909→  dispose() {
   910→    this.group.traverse((o) => {
   911→      const mesh = o as THREE.Mesh
   912→      if (mesh.geometry) mesh.geometry.dispose()
   913→      const mat = mesh.material as THREE.Material | THREE.Material[] | undefined
   914→      if (Array.isArray(mat)) mat.forEach((m) => m.dispose())
   915→      else if (mat) mat.dispose()
   916→    })
   917→    for (const d of this.disposables) d.dispose()
   918→  }
   919→}
