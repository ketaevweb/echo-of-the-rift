    16→    16→function groundMat(color: number, rough = 0.95): THREE.MeshStandardMaterial {

    20→    20→function place(group: THREE.Group, obj: THREE.Object3D, x: number, y: number, z = 0, rotY = 0) {
    21→    21→  obj.position.set(x, heightAt(x, y) + z, y)
    22→    22→  obj.rotation.y = rotY
    23→    23→  group.add(obj)
    24→    24→}

    44→    44→  // ---------------- Инстансинг: деревья/камни/кусты ----------------

    46→    46→  private buildInstanced() {
    47→    47→    const trees = PROPS.filter((p) => p.kind === 'tree')
    48→    48→    const rocks = PROPS.filter((p) => p.kind === 'rock')
    49→    49→    const shrubs = PROPS.filter((p) => p.kind === 'shrub' || p.kind === 'flower')
    50→    50→    const deadTrees = PROPS.filter((p) => p.kind === 'deadtree')
    51→    51→    const reeds = PROPS.filter((p) => p.kind === 'reed')
    52→    52→    const snowTrees = PROPS.filter((p) => p.kind === 'snowtree')
    53→    53→
    54→    54→    if (trees.length) {
    55→    55→      // Ствол
    56→    56→      const trunkGeo = new THREE.CylinderGeometry(1.6, 2.6, 14, 6)
    57→    57→      trunkGeo.translate(0, 7, 0)
    58→    58→      const trunkMat = groundMat(0x4a3626)
    59→    59→      const trunks = new THREE.InstancedMesh(trunkGeo, trunkMat, trees.length)
    60→    60→      // Крона — два конуса (матрицы инстансов — МИРОВЫЕ: база дерева + локальный сдвиг * масштаб)
    61→    61→      const coneGeo = new THREE.ConeGeometry(8, 18, 7)
    62→    62→      const leafMat = groundMat(0x2e4422)
    63→    63→      const leaves = new THREE.InstancedMesh(coneGeo, leafMat, trees.length * 2)
    64→    64→      const m = new THREE.Matrix4()
    65→    65→      const q = new THREE.Quaternion()
    66→    66→      const sc = new THREE.Vector3()
    67→    67→      trees.forEach((p, i) => {
    68→    68→        const h = heightAt(p.pos.x, p.pos.y)
    69→    69→        const rot = seedRot(p.seed ?? i)
    70→    70→        const s = 0.85 + ((p.seed ?? 7) % 100) / 220
    71→    71→        q.setFromAxisAngle(V(0, 1, 0), rot)
    72→    72→        sc.set(s, s * (0.9 + ((p.seed ?? 3) % 60) / 200), s)
    73→    73→        m.compose(V(p.pos.x, h, p.pos.y), q, sc)
    74→    74→        trunks.setMatrixAt(i, m)
    75→    75→        m.compose(V(p.pos.x, h + 12 * s, p.pos.y), q, sc)
    76→    76→        leaves.setMatrixAt(i * 2, m)
    77→    77→        m.compose(V(p.pos.x, h + 22 * s, p.pos.y), q, sc)
    78→    78→        leaves.setMatrixAt(i * 2 + 1, m)
    79→    79→      })
    80→    80→      trunks.castShadow = true
    81→    81→      leaves.castShadow = true
    82→    82→      this.group.add(trunks, leaves)
    83→    83→      this.disposables.push(trunkGeo, trunkMat, coneGeo, leafMat)
    84→    84→    }
    85→    85→
    86→    86→    if (rocks.length) {
    87→    87→      const rockGeo = new THREE.DodecahedronGeometry(1, 0)
    88→    88→      const rockMat = groundMat(0x6a6a64)
    89→    89→      const meshes = new THREE.InstancedMesh(rockGeo, rockMat, rocks.length)
    90→    90→      const m = new THREE.Matrix4()
    91→    91→      const q = new THREE.Quaternion()
    92→    92→      const e = new THREE.Euler()
    93→    93→      const sc = new THREE.Vector3()
    94→    94→      rocks.forEach((p, i) => {
    95→    95→        const h = heightAt(p.pos.x, p.pos.y)
    96→    96→        const r = p.r ?? 10
    97→    97→        e.set(seedRot(p.seed ?? i) * 0.4, seedRot((p.seed ?? 0) + 91), 0)
    98→    98→        q.setFromEuler(e)
    99→    99→        sc.set(r * 1.15, r * 0.8, r * 1.05)
   100→   100→        m.compose(V(p.pos.x, h + r * 0.25, p.pos.y), q, sc)
   101→   101→        meshes.setMatrixAt(i, m)
   102→   102→      })
   103→   103→      meshes.castShadow = true
   104→   104→      meshes.receiveShadow = true
   105→   105→      this.group.add(meshes)
   106→   106→      this.disposables.push(rockGeo, rockMat)
   107→   107→    }
   108→   108→
   109→   109→    if (shrubs.length) {
   110→   110→      const bushGeo = new THREE.IcosahedronGeometry(1, 0)
   111→   111→      const bushMat = groundMat(0x3c5226)
   112→   112→      const flowerMat = groundMat(0xa8b86a)
   113→   113→      const bushes = new THREE.InstancedMesh(bushGeo, bushMat, shrubs.length)
   114→   114→      const m = new THREE.Matrix4()
   115→   115→      const q = new THREE.Quaternion()
   116→   116→      const sc = new THREE.Vector3()
   117→   117→      shrubs.forEach((p, i) => {
   118→   118→        const h = heightAt(p.pos.x, p.pos.y)
   119→   119→        q.setFromAxisAngle(V(0, 1, 0), seedRot(p.seed ?? i))
   120→   120→        const s = p.kind === 'flower' ? 1.5 : 2.4
   121→   121→        sc.set(s, s * 0.75, s)
   122→   122→        m.compose(V(p.pos.x, h + s * 0.5, p.pos.y), q, sc)
   123→   123→        bushes.setMatrixAt(i, m)
   124→   124→      })
   125→   125→      this.group.add(bushes)
   126→   126→      this.disposables.push(bushGeo, bushMat, flowerMat)
   127→   127→    }
   128→   128→
   129→   129→    // Мёртвые деревья Вельда: голый ствол + две ветки (Фаза 3)
   130→   130→    if (deadTrees.length) {
   131→   131→      const trunkGeo = new THREE.CylinderGeometry(0.9, 2.2, 15, 6)
   132→   132→      trunkGeo.translate(0, 7.5, 0)
   133→   133→      const branchGeo = new THREE.CylinderGeometry(0.45, 0.8, 8, 5)
   134→   134→      branchGeo.translate(0, 4, 0)
   135→   135→      const barkMat = groundMat(0x3a322a, 0.98)
   136→   136→      const trunks = new THREE.InstancedMesh(trunkGeo, barkMat, deadTrees.length)
   137→   137→      const branches = new THREE.InstancedMesh(branchGeo, barkMat, deadTrees.length * 2)
   138→   138→      const m = new THREE.Matrix4()
   139→   139→      const q = new THREE.Quaternion()
   140→   140→      const q2 = new THREE.Quaternion()
   141→   141→      const e2 = new THREE.Euler()
   142→   142→      const sc = new THREE.Vector3()
   143→   143→      deadTrees.forEach((p, i) => {
   144→   144→        const h = heightAt(p.pos.x, p.pos.y)
   145→   145→        const rot = seedRot(p.seed ?? i)
   146→   146→        const s = 0.8 + ((p.seed ?? 7) % 100) / 240
   147→   147→        q.setFromAxisAngle(V(0, 1, 0), rot)
   148→   148→        sc.set(s, s, s)
   149→   149→        m.compose(V(p.pos.x, h, p.pos.y), q, sc)
   150→   150→        trunks.setMatrixAt(i, m)
   151→   151→        // Ветки: наклонены вбок от ствола
   152→   152→        e2.set(((p.seed ?? 1) % 7) * 0.05 + 0.55, rot + 1.9, 0)
   153→   153→        q2.setFromEuler(e2)
   154→   154→        m.compose(V(p.pos.x, h + 9.5 * s, p.pos.y), q2, sc)
   155→   155→        branches.setMatrixAt(i * 2, m)
   156→   156→        e2.set(0.7, rot + 4.6, 0)
   157→   157→        q2.setFromEuler(e2)
   158→   158→        m.compose(V(p.pos.x, h + 7.5 * s, p.pos.y), q2, sc)
   159→   159→        branches.setMatrixAt(i * 2 + 1, m)
   160→   160→      })
   161→   161→      trunks.castShadow = true
   162→   162→      branches.castShadow = true
   163→   163→      this.group.add(trunks, branches)
   164→   164→      this.disposables.push(trunkGeo, branchGeo, barkMat)
   165→   165→    }
   166→   166→
   167→   167→    // Камыш у омутов (Фаза 3)
   168→   168→    if (reeds.length) {
   169→   169→      const reedGeo = new THREE.ConeGeometry(0.9, 6.5, 4)
   170→   170→      reedGeo.translate(0, 3.2, 0)
   171→   171→      const reedMat = groundMat(0x4a5530, 0.98)
   172→   172→      const mesh = new THREE.InstancedMesh(reedGeo, reedMat, reeds.length * 3)
   173→   173→      const m = new THREE.Matrix4()
   174→   174→      const q = new THREE.Quaternion()
   175→   175→      const sc = new THREE.Vector3()
   176→   176→      reeds.forEach((p, i) => {
   177→   177→        const h = heightAt(p.pos.x, p.pos.y)
   178→   178→        for (let k = 0; k < 3; k++) {
   179→   179→          const ox = ((p.seed ?? i) % 13) - 6 + k * 3.1
   180→   180→          const oz = (((p.seed ?? i) * 7) % 11) - 5
   181→   181→          q.setFromAxisAngle(V(0, 1, 0), seedRot((p.seed ?? 0) + k * 31))
   182→   182→          const s = 0.7 + ((p.seed ?? 3) % 50) / 90
   183→   183→          sc.set(s, s * (0.85 + k * 0.12), s)
   184→   184→          m.compose(V(p.pos.x + ox, h, p.pos.y + oz), q, sc)
   185→   185→          mesh.setMatrixAt(i * 3 + k, m)
   186→   186→        }
   187→   187→      })
   188→   188→      this.group.add(mesh)
   189→   189→      this.disposables.push(reedGeo, reedMat)
   190→   190→    }
   191→   191→
   192→   192→    // Снежные ели Пределья (Фаза 6.2): тёмная хвоя под шапкой снега
   193→   193→    if (snowTrees.length) {
   194→   194→      const trunkGeo = new THREE.CylinderGeometry(1.3, 2.2, 12, 6)
   195→   195→      trunkGeo.translate(0, 6, 0)
   196→   196→      const barkMat = groundMat(0x3c3226)
   197→   197→      const trunks = new THREE.InstancedMesh(trunkGeo, barkMat, snowTrees.length)
   198→   198→      const coneGeo = new THREE.ConeGeometry(6.5, 15, 7)
   199→   199→      const needleMat = groundMat(0x2a3a28)
   200→   200→      const snowMatI = groundMat(0xd8dee0, 0.92)
   201→   201→      const cones = new THREE.InstancedMesh(coneGeo, needleMat, snowTrees.length * 2)
   202→   202→      const caps = new THREE.InstancedMesh(new THREE.ConeGeometry(4.4, 5.5, 7), snowMatI, snowTrees.length)
   203→   203→      const m = new THREE.Matrix4()
   204→   204→      const q = new THREE.Quaternion()
   205→   205→      const sc = new THREE.Vector3()
   206→   206→      snowTrees.forEach((p, i) => {
   207→   207→        const h = heightAt(p.pos.x, p.pos.y)
   208→   208→        const rot = seedRot(p.seed ?? i)
   209→   209→        const s = 0.85 + ((p.seed ?? 7) % 100) / 200
   210→   210→        q.setFromAxisAngle(V(0, 1, 0), rot)
   211→   211→        sc.set(s, s, s)
   212→   212→        m.compose(V(p.pos.x, h, p.pos.y), q, sc)
   213→   213→        trunks.setMatrixAt(i, m)
   214→   214→        m.compose(V(p.pos.x, h + 11 * s, p.pos.y), q, sc)
   215→   215→        cones.setMatrixAt(i * 2, m)
   216→   216→        m.compose(V(p.pos.x, h + 19 * s, p.pos.y), q, sc)
   217→   217→        cones.setMatrixAt(i * 2 + 1, m)
   218→   218→        m.compose(V(p.pos.x, h + 26.5 * s, p.pos.y), q, sc)
   219→   219→        caps.setMatrixAt(i, m)
   220→   220→      })
   221→   221→      trunks.castShadow = true
   222→   222→      cones.castShadow = true
   223→   223→      this.group.add(trunks, cones, caps)
   224→   224→      this.disposables.push(trunkGeo, barkMat, coneGeo, needleMat, snowMatI)
   225→   225→    }
   226→   226→  }

   230→   230→  private buildProp(p: PropDef) {
   231→   231→    const { x, y } = p.pos
   232→   232→    const h = () => heightAt(x, y)
   233→   233→    switch (p.kind) {
   234→   234→      case 'tree':
   235→   235→      case 'rock':
   236→   236→      case 'shrub':
   237→   237→      case 'flower':
   238→   238→        return // уже в инстансах
   239→   239→
   240→   240→      case 'house': {
   241→   241→        const g = new THREE.Group()
   242→   242→        const w = p.w ?? 120
   243→   243→        const d = p.h ?? 70
   244→   244→        const bodyH = 26 + w * 0.06
   245→   245→        const body = new THREE.Mesh(new THREE.BoxGeometry(w, bodyH, d), groundMat(0x6b543c))
   246→   246→        body.position.y = bodyH / 2
   247→   247→        body.castShadow = true
   248→   248→        body.receiveShadow = true
   249→   249→        const roof = new THREE.Mesh(new THREE.ConeGeometry(Math.max(w, d) * 0.62, 20 + w * 0.08, 4), groundMat(0x3d2f22))
   250→   250→        roof.rotation.y = Math.PI / 4
   251→   251→        roof.position.y = bodyH + (10 + w * 0.04)
   252→   252→        roof.castShadow = true
   253→   253→        g.add(body, roof)
   254→   254→        place(this.group, g, x, y, 0, seedRot(p.seed ?? 1) * 0.2)
   255→   255→        break
   256→   256→      }
   257→   257→
   258→   258→      case 'tent': {
   259→   259→        const g = new THREE.Group()
   260→   260→        const w = p.w ?? 60
   261→   261→        const tent = new THREE.Mesh(new THREE.ConeGeometry(w * 0.42, w * 0.62, 5), groundMat(0x7a6248))
   262→   262→        tent.position.y = w * 0.31
   263→   263→        tent.castShadow = true
   264→   264→        g.add(tent)
   265→   265→        place(this.group, g, x, y, 0, seedRot(p.seed ?? 1))
   266→   266→        break
   267→   267→      }
   268→   268→
   269→   269→      case 'stall': {
   270→   270→        const g = new THREE.Group()
   271→   271→        const table = new THREE.Mesh(new THREE.BoxGeometry(70, 8, 42), groundMat(0x6b543c))
   272→   272→        table.position.y = 16
   273→   273→        const roof = new THREE.Mesh(new THREE.BoxGeometry(80, 4, 50), groundMat(0x8a3d2e))
   274→   274→        roof.position.y = 40
   275→   275→        roof.rotation.z = 0.08
   276→   276→        for (const [lx, lz] of [[-30, -16], [30, -16], [-30, 16], [30, 16]]) {
   277→   277→          const leg = new THREE.Mesh(new THREE.CylinderGeometry(1.6, 1.6, 32, 5), groundMat(0x4a3626))
   278→   278→          leg.position.set(lx, 16, lz)
   279→   279→          g.add(leg)
   280→   280→        }
   281→   281→        table.castShadow = true
   282→   282→        g.add(table, roof)
   283→   283→        place(this.group, g, x, y)
   284→   284→        break
   285→   285→      }
   286→   286→
   287→   287→      case 'dummy': {
   288→   288→        const g = new THREE.Group()
   289→   289→        const post = new THREE.Mesh(new THREE.CylinderGeometry(3, 4, 34, 6), groundMat(0x5c4630))
   290→   290→        post.position.y = 17
   291→   291→        const cross = new THREE.Mesh(new THREE.BoxGeometry(26, 4, 4), groundMat(0x5c4630))
   292→   292→        cross.position.y = 26
   293→   293→        const head = new THREE.Mesh(new THREE.SphereGeometry(4.5, 8, 6), groundMat(0x8a7148))
   294→   294→        head.position.y = 37
   295→   295→        post.castShadow = true
   296→   296→        g.add(post, cross, head)
   297→   297→        place(this.group, g, x, y)
   298→   298→        break
   299→   299→      }
   300→   300→
   301→   301→      case 'fire': {
   302→   302→        const g = new THREE.Group()
   303→   303→        for (let i = 0; i < 5; i++) {
   304→   304→          const log = new THREE.Mesh(new THREE.CylinderGeometry(1.4, 1.8, 12, 5), groundMat(0x3d2a1a))
   305→   305→          log.rotation.set(Math.PI / 2.4, (i / 5) * Math.PI * 2, 0)
   306→   306→          log.position.y = 2.4
   307→   307→          g.add(log)
   308→   308→        }
   309→   309→        const flame = new THREE.Mesh(
   310→   310→          new THREE.ConeGeometry(5, 12, 6),
   311→   311→          new THREE.MeshBasicMaterial({ color: 0xe8913a, transparent: true, opacity: 0.92 }),
   312→   312→        )
   313→   313→        flame.position.y = 9
   314→   314→        const light = new THREE.PointLight(0xff9540, 320, 190, 2)
   315→   315→        light.position.y = 12
   316→   316→        g.add(flame, light)
   317→   317→        place(this.group, g, x, y)
   318→   318→        this.fires.push({ light, flame, base: 1 })
   319→   319→        break
   320→   320→      }
   321→   321→
   322→   322→      case 'fence': {
   323→   323→        const g = new THREE.Group()
   324→   324→        const w = p.w ?? 90
   325→   325→        const rail = new THREE.Mesh(new THREE.BoxGeometry(w, 3, 2.4), groundMat(0x54422e))
   326→   326→        rail.position.y = 12
   327→   327→        const rail2 = rail.clone()
   328→   328→        rail2.position.y = 6
   329→   329→        g.add(rail, rail2)
   330→   330→        const n = Math.max(2, Math.round(w / 30))
   331→   331→        for (let i = 0; i <= n; i++) {
   332→   332→          const post = new THREE.Mesh(new THREE.CylinderGeometry(1.6, 1.6, 14, 5), groundMat(0x4a3626))
   333→   333→          post.position.set(-w / 2 + (w / n) * i, 7, 0)
   334→   334→          g.add(post)
   335→   335→        }
   336→   336→        place(this.group, g, x, y, 0, seedRot(p.seed ?? 1) * 0.1)
   337→   337→        break
   338→   338→      }
   339→   339→
   340→   340→      case 'stone': {
   341→   341→        const g = new THREE.Group()
   342→   342→        const r = p.r ?? 12
   343→   343→        const obelisk = new THREE.Mesh(new THREE.BoxGeometry(r * 0.9, r * 3.4, r * 0.7), groundMat(0x77776f))
   344→   344→        obelisk.position.y = r * 1.6
   345→   345→        obelisk.rotation.z = (seedRot(p.seed ?? 1) - Math.PI) * 0.06
   346→   346→        obelisk.castShadow = true
   347→   347→        const cap = new THREE.Mesh(new THREE.ConeGeometry(r * 0.55, r * 0.9, 4), groundMat(0x6a6a62))
   348→   348→        cap.position.y = r * 3.4
   349→   349→        cap.rotation.y = Math.PI / 4
   350→   350→        g.add(obelisk, cap)
   351→   351→        place(this.group, g, x, y)
   352→   352→        break
   353→   353→      }
   354→   354→
   355→   355→      case 'altar': {
   356→   356→        const g = new THREE.Group()
   357→   357→        const slab = new THREE.Mesh(new THREE.CylinderGeometry(16, 20, 8, 8), groundMat(0x8a8a80))
   358→   358→        slab.position.y = 4
   359→   359→        slab.castShadow = true
   360→   360→        const glow = new THREE.Mesh(
   361→   361→          new THREE.OctahedronGeometry(5, 0),
   362→   362→          new THREE.MeshBasicMaterial({ color: 0xd9a441, transparent: true, opacity: 0.85 }),
   363→   363→        )
   364→   364→        glow.position.y = 22
   365→   365→        const ring = new THREE.Mesh(
   366→   366→          new THREE.RingGeometry(20, 23, 32),
   367→   367→          new THREE.MeshBasicMaterial({ color: 0xd9a441, transparent: true, opacity: 0.3, side: THREE.DoubleSide }),

   872→   872→  update(time: number, removed: Set<string>, darkness = 0, runes?: { rs1: number; rs2: number; rs3: number }) {
   873→   873→    // Огни костров мерцают; ночью — ярче (Фаза 3)
   874→   874→    const nightBoost = 0.8 + darkness * 1.5
   875→   875→    for (const f of this.fires) {
   876→   876→      const flick = 0.82 + Math.sin(time * 11) * 0.1 + Math.sin(time * 23.7) * 0.08
   877→   877→      f.light.intensity = 320 * flick * nightBoost
   878→   878→      const s = 0.9 + flick * 0.25
   879→   879→      f.flame.scale.set(s, 1 + flick * 0.4, s)
   880→   880→    }
   881→   881→    // Пульс колец трав/алтаря
   882→   882→    const pulse = 0.5 + 0.5 * Math.sin(time * 2.5)
   883→   883→    for (const r of this.herbPulse) {
   884→   884→      const m = (r as THREE.Mesh).material as THREE.MeshBasicMaterial
   885→   885→      m.opacity = 0.22 + pulse * 0.3
   886→   886→    }
   887→   887→    // Вращение магических элементов
   888→   888→    for (const s of this.spinners) {
   889→   889→      s.rotation.y = time * 0.9
   890→   890→      s.rotation.z = Math.sin(time * 0.6) * 0.15
   891→   891→    }
   892→   892→    // Фаза 6.2: кольца рунных камней перекрашиваются по состоянию головоломки
   893→   893→    if (runes) {
   894→   894→      const colors = [0x9fb8a4, 0xd9a441, 0xd8dde2]
   895→   895→      for (const [id, mesh] of this.runeRings) {
   896→   896→        const idx = runes[id as 'rs1'] ?? 0
   897→   897→        const mat = mesh.material as THREE.MeshStandardMaterial
   898→   898→        mat.color.setHex(colors[idx])
   899→   899→        mat.emissive.setHex(colors[idx])
   900→   900→      }
   901→   901→    }
   902→   902→    // Скрытые пропсы (барьеры по квесту, собранные травы)
   903→   903→    for (const [id, objs] of this.removable) {
   904→   904→      const hidden = removed.has(id)
   905→   905→      for (const o of objs) o.visible = !hidden
   906→   906→    }
   907→   907→  }
   908→   908→
   909→   909→  dispose() {
   910→   910→    this.group.traverse((o) => {
   911→   911→      const mesh = o as THREE.Mesh
   912→   912→      if (mesh.geometry) mesh.geometry.dispose()
   913→   913→      const mat = mesh.material as THREE.Material | THREE.Material[] | undefined
   914→   914→      if (Array.isArray(mat)) mat.forEach((m) => m.dispose())
   915→   915→      else if (mat) mat.dispose()
   916→   916→    })
   917→   917→    for (const d of this.disposables) d.dispose()
   918→   918→  }
   919→   919→}
